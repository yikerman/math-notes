The mathematical framework for probability involves set theory.

\subsection{Sample Spaces and Events}

\begin{example}[Sample Spaces]
  \begin{itemize}
    \item Toss a coin: $S=\{\text{Head},\text{Tail}\}$.
    \item Toss two coins: $S=\{(H,H),(H,T),(T,H),(T,T)\}$.
    \item Roll a die: $S=\{1,2,3,4,5,6\}$.
    \item Permute $1$--$6$: $S=\{123456,213456,312456,\ldots\}$.
  \end{itemize}
\end{example}

\begin{definition}{Outcomes and Events}{outcomes-events}
  An element of a sample space is an \textbf{outcome}.
  A subset of a sample space is an \textbf{event}.
\end{definition}

\begin{example}[Sum of Two Dice]
  \[
    E=\{\text{sum of two dice rolls is }7\}
    =\{(1,6),(2,5),\ldots,(6,1)\}\subseteq S,
  \]
  where $S$ is the sample space of two dice rolls.
\end{example}

\subsection{Probability}

Write $EF=E\cap F$ for intersection.

\begin{definition}{Probability}{probability}
  A probability on a sample space $S$ is a function $P$ defined on subsets of $S$ with properties:
  \begin{enumerate}
    \item $\forall E\subseteq S,\quad 0\le P(E)\le1$.
    \item $E,F\subseteq S$ and $EF=\varnothing$ imply
          \[
            P(E\cup F)=P(E)+P(F).
          \]
    \item $P(S)=1$.
  \end{enumerate}
\end{definition}

\begin{example}[Comparing Two Dice Rolls]
  \begin{align*}
    S & =\{(1,1),(1,2),\ldots,\text{all two-dice rolls}\},              \\
    E & =\{\text{outcome of first roll}>\text{outcome of second roll}\} \\
      & =\{(2,1),(3,1),(3,2),\ldots\}.
  \end{align*}
  By property (2) of Definition~\ref{def:probability},
  \[
    P(E)=P(\{(2,1)\})+P(\{(3,1)\})+P(\{(3,2)\})+\cdots.
  \]
\end{example}

\subsection{Some Properties}

\begin{theorem}{Properties of Probability}{probability-properties}
  \begin{enumerate}
    \item $P(\varnothing)=0$.
    \item $P(E)+P(E^c)=1$, where $E^c=S\setminus E$.
    \item $P(A\cup B)=P(A)+P(B)-P(AB)$.
  \end{enumerate}
\end{theorem}

\begin{proof}
  \begin{enumerate}
    \item Since $S\varnothing=\varnothing$ and $S\cup\varnothing=S$,
          \[
            P(\varnothing)+P(S)=P(S)\implies P(\varnothing)=0.
          \]
    \item Since $EE^c=\varnothing$ and $E\cup E^c=S$,
          \[
            P(E)+P(E^c)=P(S)=1.
          \]
    \item Since $A\cup B=A\cup(BA^c)$,
          \begin{center}
            \begin{tikzpicture}
              \draw (0,0) circle (0.8);
              \draw (1,0) circle (0.8);
              \node at (-0.3,-1.1) {$A$};
              \node at (1.3,-1.1) {$B$};
            \end{tikzpicture}
          \end{center}
          \[
            P(A\cup B)=P(A\cup BA^c)=P(A)+P(BA^c).
          \]
          Also, $BA\cap BA^c=\varnothing$ and $BA\cup BA^c=B$, so
          \[
            P(B)=P(BA)+P(BA^c).
          \]
          Hence $P(A\cup B)=P(A)+P(B)-P(AB)$.\qedhere
  \end{enumerate}
\end{proof}

\begin{theorem}{Inclusion--Exclusion}{inclusion-exclusion}
  Generalizing property (3) of Theorem~\ref{thm:probability-properties},
  \begin{align*}
    P\left(\bigcup_{i=1}^n E_i\right)
     & =\sum_{i=1}^n P(E_i)-\sum_{1\le i<j\le n}P(E_iE_j) \\
     & \quad+\sum_{1\le i<j<k\le n}P(E_iE_jE_k)-\cdots    \\
     & \quad+(-1)^{n+1}P(E_1E_2\cdots E_n).
  \end{align*}
\end{theorem}

\subsection{Equally Likely Outcomes}

Let $\#S$ denote the number of elements in $S$. We usually have $\#S<\infty$.
Many times it is reasonable to consider all outcomes equally likely. Then
\[
  E\subseteq S\implies P(E)=\frac{\#E}{\#S}.
\]

\begin{example}[Poker: Flush]
  $E=\{\text{all cards in the same suit}\}$, and $S$ is the set of all possible five-card hands.
  \[
    \#S=\binom{52}{5},\qquad \#E=4\binom{13}{5}.
  \]
  Thus
  \[
    P(\text{flush})=\frac{4\binom{13}{5}}{\binom{52}{5}}
    =\frac{4\cdot13!\cdot47!}{8!\cdot52!}.
  \]
\end{example}

\begin{example}[Red and Blue Balls]
  Order $m$ red balls and $n$ blue balls in a row. How many different color arrangements are there?

  Label the balls $r_1,r_2,\ldots,r_m$ and $b_1,b_2,\ldots,b_n$.
  There are $(m+n)!$ ways to arrange them, and
  \[
    P(\text{a color arrangement})=\frac{m!\,n!}{(m+n)!}.
  \]
\end{example}

\subsection{Set Identities}

\begin{theorem}{Distributive Laws}{set-distributive-laws}
  \[
    \left(\bigcup_{i=1}^{\infty}E_i\right)F
    =\bigcup_{i=1}^{\infty}E_iF,
    \qquad
    \left(\bigcap_{i=1}^{\infty}E_i\right)\cup F
    =\bigcap_{i=1}^{\infty}(E_i\cup F).
  \]
\end{theorem}

\begin{proof}
  For the first identity:
  \begin{enumerate}[label=(\roman*)]
    \item Suppose $x\in\left(\bigcup_{i=1}^{\infty}E_i\right)F$.
          Then $x\in\bigcup_{i=1}^{\infty}E_i$ and $x\in F$.
          Thus there exists $i$ with $x\in E_i$ and $x\in F$, so
          \[
            x\in E_iF\subseteq\bigcup_{i=1}^{\infty}E_iF.
          \]
          Hence $\left(\bigcup_{i=1}^{\infty}E_i\right)F
            \subseteq\bigcup_{i=1}^{\infty}E_iF$.
    \item Suppose $x\in\bigcup_{i=1}^{\infty}E_iF$.
          Then there exists $i$ with $x\in E_iF$, so $x\in E_i$ and $x\in F$.
          Since $E_i\subseteq\bigcup_{i=1}^{\infty}E_i$,
          \[
            x\in\bigcup_{i=1}^{\infty}E_i\quad\text{and}\quad x\in F
            \implies x\in\left(\bigcup_{i=1}^{\infty}E_i\right)F.
          \]
          Hence $\bigcup_{i=1}^{\infty}E_iF
            \subseteq\left(\bigcup_{i=1}^{\infty}E_i\right)F$.
  \end{enumerate}
  For the second identity:
  \begin{enumerate}[label=(\roman*)]
    \item Suppose $x\in\left(\bigcap_{i=1}^{\infty}E_i\right)\cup F$.
          Then $x\in\bigcap_{i=1}^{\infty}E_i$ or $x\in F$.
          Thus, for every $i$, $x\in E_i\cup F$, so
          \[
            x\in\bigcap_{i=1}^{\infty}(E_i\cup F).
          \]
          Hence $\left(\bigcap_{i=1}^{\infty}E_i\right)\cup F
            \subseteq\bigcap_{i=1}^{\infty}(E_i\cup F)$.
    \item Vice versa.\qedhere
  \end{enumerate}
\end{proof}

\begin{theorem}{De Morgan's Law}{de-morgan-law}
  \[
    (A\cup B)^c=A^c\cap B^c.
  \]
  More generally,
  \[
    \left(\bigcup_{i=1}^n E_i\right)^c
    =\bigcap_{i=1}^n E_i^c.
  \]
\end{theorem}

\begin{center}
  \begin{tikzpicture}
    \fill[gray!20] (-1.2,-1.1) rectangle (2.2,1.1);
    \fill[white] (0,0) circle (0.8);
    \fill[white] (1,0) circle (0.8);
    \draw (0,0) circle (0.8);
    \draw (1,0) circle (0.8);
    \node at (-0.3,-0.4) {$A$};
    \node at (1.3,-0.4) {$B$};
  \end{tikzpicture}
\end{center}

\begin{proof}
  \begin{align*}
    x\in(A\cup B)^c
     & \iff x\notin A\cup B                \\
     & \iff x\notin A\text{ and }x\notin B \\
     & \iff x\in A^c\text{ and }x\in B^c   \\
     & \iff x\in A^c\cap B^c.
  \end{align*}
  For the general identity, use induction on $n$.
\end{proof}

\subsection{Birthday Paradox and Random Matching}

\begin{example}[Birthday Paradox]
  In a group of $n$ people, what is the probability that no two people have the same birthday?
  \[
    \#S=365^n,\qquad
    \#\{\text{no overlap}\}
    =365\cdot364\cdots(365-(n-1))
    =\prod_{i=0}^{n-1}(365-i).
  \]
  Thus
  \[
    p=\frac{\#\{\text{no overlap}\}}{\#S}
    =\prod_{i=0}^{n-1}\frac{365-i}{365}
    =\prod_{i=0}^{n-1}\left(1-\frac{i}{365}\right).
  \]
  This gets small fast: when $n=23$, $p<\frac12$.
\end{example}

\begin{example}[Drawing Keys: No One Gets the Right Key]
  $n$ people put their keys in a basket and draw them out randomly.
  What is the probability that no one gets the right key?

  Let $E_i=\{\text{person }i\text{ gets the right key}\}$.
  By Theorem~\ref{thm:de-morgan-law},
  \[
    P\left(\bigcap_{i=1}^n E_i^c\right)
    =P\left(\left(\bigcup_{i=1}^n E_i\right)^c\right)
    =1-P\left(\bigcup_{i=1}^n E_i\right).
  \]
  No one gets it right $=1-\text{some get it right}$.

  By Theorem~\ref{thm:inclusion-exclusion},
  \[
    P\left(\bigcup_{i=1}^n E_i\right)
    =\sum_{i=1}^n P(E_i)-\sum_{i<j}P(E_iE_j)
    +\sum_{i<j<k}P(E_iE_jE_k)-\cdots.
  \]
  There are $n!$ equally likely assignments of keys to people.
  For $E_i$, person $i$ receives their own key in one way, and the remaining
  $n-1$ keys can be assigned in $(n-1)!$ ways. Thus
  \[
    P(E_i)=\frac{1\cdot(n-1)!}{n!}=\frac1n.
  \]
  For distinct $i,j$, $E_iE_j$ means both people $i,j$ get their own keys.
  Each of these two keys is fixed in one way. There is no constraint on the
  remaining $n-2$ people: they may get their own keys or someone else's.
  Their keys can therefore be assigned in $(n-2)!$ ways, giving
  \[
    P(E_iE_j)=\frac{1\cdot1\cdot(n-2)!}{n!}
    =\frac{(n-2)!}{n(n-1)(n-2)!}
    =\frac{1}{n(n-1)}.
  \]
  Similarly, fixing the keys of three distinct people leaves $(n-3)!$ assignments:
  \[
    P(E_iE_jE_k)=\frac{(n-3)!}{n!}
    =\frac{1}{n(n-1)(n-2)}.
  \]
  There are $n$ single events, $\binom n2$ pairs, $\binom n3$ triples, and so on.
  Hence
  \begin{align*}
    P\left(\bigcup_{i=1}^n E_i\right)
     & =n\cdot\frac1n-\binom n2\frac{1}{n(n-1)}
    +\binom n3\frac{1}{n(n-1)(n-2)}-\cdots        \\
     & =1-\frac1{2!}+\frac1{3!}-\frac1{4!}+\cdots
    =\sum_{i=1}^n\frac{(-1)^{i+1}}{i!}.
  \end{align*}
  Therefore
  \begin{align*}
    P\left(\bigcap_{i=1}^n E_i^c\right)
     & =1-P\left(\bigcup_{i=1}^n E_i\right) \\
     & =1-1+\frac1{2!}-\frac1{3!}+\cdots
    =\sum_{k=0}^n\frac{(-1)^k}{k!}.
  \end{align*}
  Recall $e^x=\sum_{m=0}^{\infty}\frac{x^m}{m!}$. As $n\to\infty$,
  \[
    P\left(\bigcap_{i=1}^n E_i^c\right)\longrightarrow e^{-1}.
  \]

\end{example}

\begin{example}[Drawing Keys: Exactly $k$ People Get the Right Key]
  Let $r_n$ be the number of ways no one gets the right key, so
  $r_n/n!=P\left(\bigcap_{i=1}^n E_i^c\right)$.
  Choose the one person who gets it right in $\binom n1$ ways;
  the remaining people must all get wrong keys, in $r_{n-1}$ ways. Then
  \[
    P(\text{exactly one gets the right key})
    =\frac{\binom n1 r_{n-1}}{n!}
    =\frac{r_{n-1}}{(n-1)!}.
  \]
  As $n\to\infty$,
  \[
    P(\text{exactly one gets the right key})\longrightarrow e^{-1}.
  \]
  More generally, choose the $k$ people who get their own keys in $\binom nk$ ways.
  The remaining $n-k$ people must all get wrong keys, in $r_{n-k}$ ways. Thus
  \begin{align*}
    P(\text{exactly }k\text{ get the right key})
     & =\frac{\binom nk r_{n-k}}{n!}
    =\frac{n!\,r_{n-k}}{k!\,(n-k)!\,n!}   \\
     & =\frac1{k!}\frac{r_{n-k}}{(n-k)!}.
  \end{align*}
  As $n\to\infty$ with $k$ fixed,
  \[
    P(\text{exactly }k\text{ get the right key})
    \longrightarrow\frac{e^{-1}}{k!}.
  \]
\end{example}
