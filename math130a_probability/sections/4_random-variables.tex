\subsection{Random Variables and Probability Mass Functions}

\begin{definition}{Random Variable}{random-variable}
  A random variable on a sample space $S$ is a function $X:S\to\R$.
\end{definition}

\begin{example}
  $S=\{THT\cdots,\ldots\}$, the sample space of tossing a coin $n$ times.
  For $\omega\in S$,
  \[
    X(\omega)=\#\text{ H's in }\omega
  \]
  is a random variable.

  Another random variable:
  \[
    X(\omega)=
    \begin{cases}
      \text{position of first H in }\omega, & \text{if there is an H}, \\
      n+1,                                  & \text{if no H}.
    \end{cases}
  \]
\end{example}

\begin{example}
  $S=\{(1,1),(1,2),\ldots,(6,6)\}$.
  For $\omega=(\omega_1,\omega_2)\in S$,
  \[
    X(\omega)=\omega_1+\omega_2.
  \]
\end{example}

\begin{example}[Largest Number Selected]
  20 numbered balls in an urn; choose 4 at random.
  \[
    S=\bigl\{\{1,2,3,4\},\{1,2,3,5\},\ldots,\{17,18,19,20\}\bigr\}.
  \]
  $X(\omega)=$ largest number selected.
  \begin{gather*}
    P(X=4)=\frac1{\binom{20}{4}},\qquad
    P(X=5)=\frac{\binom43}{\binom{20}{4}},\\
    P(X=6)=\frac{\binom53}{\binom{20}{4}},\qquad\ldots,\qquad
    P(X=k)=\frac{\binom{k-1}{3}}{\binom{20}{4}},\quad k=4,\ldots,20.
  \end{gather*}
  \[
    \sum_{k=4}^{20}P(X=k)=1\qquad\text{(all of sample space)}.
  \]
\end{example}

\begin{definition}{Probability Mass Function}{probability-mass-function}
  For a discrete random variable $X$,
  \[
    p(x)=P(X=x)
  \]
  is the probability mass function (pmf) of $X$.
\end{definition}

For a random variable, we have $E=\{\omega:X(\omega)=x\}\subseteq S$,
\[
  P(E)=P(\{\omega:X(\omega)=x\})=P(X=x)=p(x).
\]
Properties:
\[
  p(x)\ge0,\qquad\sum_{x:p(x)>0}p(x)=1.
\]
\[
  P(X\in E)=\sum_{x\in E}p(x).
\]

\begin{definition}{Cumulative Distribution Function}{cumulative-distribution-function}
  $F(x)=P(X\le x)$ is called the cumulative distribution function of $X$.
\end{definition}

\subsection{Discrete Distributions}

\begin{definition}{Uniform Distribution}{discrete-uniform-distribution}
  Uniform on $\{1,2,\ldots,n\}$:
  \[
    p(x)=P(X=x)=\frac1n,\qquad x=1,\ldots,n.
  \]
\end{definition}

\begin{example}[Toss Until Heads]
  Toss a coin until H appears.
  \[
    S=\{H,TH,TTH,\ldots\},\qquad P(H)=p\in(0,1).
  \]
  $X(\omega)=$ length of $\omega$.
  \[
    \{\omega:X(\omega)=k\}=\{X=k\}.
  \]
  \begin{align*}
    P(X=1) & =p,            \\
    P(X=2) & =(1-p)p,       \\
    P(X=3) & =(1-p)^2p,     \\
    P(X=k) & =(1-p)^{k-1}p.
  \end{align*}
\end{example}

\begin{definition}{Geometric Distribution}{geometric-distribution}
  Geometric pmf with parameter $p\in(0,1)$:
  \[
    P(X=x)=p(x)=(1-p)^{x-1}p,\qquad x=1,2,\ldots.
  \]
  First success on $x$.\\
  A random variable with geometric pmf is a geometric random variable.
\end{definition}

\begin{gather*}
  \sum_{k=1}^{\infty}p(1-p)^{k-1}\stackrel{?}{=}1,\\
  \text{WTS }p\sum_{k=1}^{\infty}(1-p)^{k-1}=1,\\
  \text{WTS }p\sum_{\ell=0}^{\infty}(1-p)^\ell=1.
\end{gather*}
Geometric series:
\[
  |r|<1\implies\sum_{k=0}^{\infty}r^k=\frac1{1-r}.
\]

\begin{example}[Number of Heads]
  Toss a coin $n$ times (tosses independent). $S$ is the sample space:
  \[
    S=\{HH\cdots H,HTH\cdots H,HHTH\cdots H,\ldots,TT\cdots T\}.
  \]
  An $\omega\in S$ is a sequence of length $n$,
  \[
    \omega=(\omega_1,\omega_2,\ldots,\omega_n),\qquad\omega_i\in\{H,T\}.
  \]
  \[
    X(\omega)=\#\{i:\omega_i=H\}.
  \]
  $P(X=k)$: probability that the number of H's is $k$.

  For $n=5$, $k=3$:
  \[
    \begin{array}{ccccc}
      H & H & H & T & T \\
      H & H & T & H & T \\
      \multicolumn{5}{c}{\cdots}
    \end{array}
  \]
  \[
    P(X=k)=\binom nk p^k(1-p)^{n-k}.
  \]
\end{example}

\begin{definition}{Binomial Distribution}{binomial-distribution}
  Binomial pmf:
  \[
    p(k)=\binom nk p^k(1-p)^{n-k},\qquad k=0,\ldots,n.
  \]
  Written as $\operatorname{Bi}(n,p)$, with success probability $p$, $n$ trials.
\end{definition}

\begin{definition}{Poisson Distribution}{poisson-distribution}
  $X$ is called a Poisson random variable with parameter $\lambda>0$ if
  \[
    P(X=x)=\frac{\lambda^x}{x!}e^{-\lambda}=p(x),\qquad x=0,1,2,\ldots.
  \]
\end{definition}

Note: for the Poisson pmf,
\[
  \sum_{m=0}^{\infty}\frac{\lambda^m}{m!}e^{-\lambda}
  =e^{-\lambda}\sum_{m=0}^{\infty}\frac{\lambda^m}{m!}
  =e^{-\lambda}e^\lambda=1.
\]

\subsubsection{Binomial to Poisson}

Note that
\begin{align*}
  \text{Poisson r.v.: }  & p(k)=\frac{\lambda^k}{k!}e^{-\lambda}, \\
  \text{Binomial r.v.: } & p(k)=\binom nk p^k(1-p)^{n-k}.
\end{align*}
For $Y\sim\operatorname{Bi}(n,p)$, take $p=\lambda/n$ and
$Y_n\sim\operatorname{Bi}(n,\lambda/n)$.
\begin{align*}
  P(Y_n=0) & =\left(1-\frac\lambda n\right)^n\to e^{-\lambda},           \\
  P(Y_n=1) & =n\frac\lambda n\left(1-\frac\lambda n\right)^{n-1}
  \to\lambda e^{-\lambda},                                               \\
  P(Y_n=2) & =\frac{n(n-1)}2\frac{\lambda^2}{n^2}
  \left(1-\frac\lambda n\right)^{n-2}\to\frac{\lambda^2}{2}e^{-\lambda}, \\
  P(Y_n=3) & =\frac{n(n-1)(n-2)}{3\cdot2\cdot1}\frac{\lambda^3}{n^3}
  \left(1-\frac\lambda n\right)^{n-3}\to\frac{\lambda^3}{3!}e^{-\lambda}.
\end{align*}
Poisson distribution.

\subsection{Expected Value}

\begin{definition}{Expected Value}{expected-value}
  For a random variable $X$ with pmf $p$ and $\sum_x|x|p(x)<\infty$,
  \[
    EX=\sum_x xp(x)
  \]
  is the expected value of $X$.
\end{definition}

\begin{example}[Uniform]
  \[
    EX=\sum_x xp(x)=\sum_{k=1}^n k\cdot\frac1n
    =\frac1n\cdot\frac{n(n+1)}2=\frac{n+1}2.
  \]
\end{example}

\begin{example}[Binomial]
  For $\operatorname{Bi}(n,p)$:
  \begin{align*}
    EX & =\sum_{k=0}^n kp(k)=\sum_{k=0}^n k\binom nk p^k(1-p)^{n-k}       \\
       & =\sum_{k=0}^n k\frac{n!}{k!(n-k)!}p^k(1-p)^{n-k}                 \\
       & =\sum_{k=1}^n\frac{n!}{(k-1)!(n-k)!}p^k(1-p)^{n-k}               \\
       & =np\sum_{k=1}^n\frac{(n-1)!}{(k-1)!(n-1-(k-1))!}
    p^{k-1}(1-p)^{n-1-(k-1)}                                              \\
       & =np\sum_{j=0}^{n-1}\frac{(n-1)!}{j!((n-1)-j)!}p^j(1-p)^{(n-1)-j} \\
       & =np\sum_{j=0}^{n-1}\binom{n-1}{j}p^j(1-p)^{n-1-j}                \\
       & =np[p+(1-p)]^{n-1}=np.
  \end{align*}
\end{example}

\begin{example}[Poisson]
  For $\operatorname{Po}(\lambda)$:
  \begin{align*}
    EX & =\sum_{k=0}^{\infty}k\frac{\lambda^k}{k!}e^{-\lambda}
    =e^{-\lambda}\sum_{k=0}^{\infty}k\frac{\lambda^k}{k!}                      \\
       & =e^{-\lambda}\sum_{k=1}^{\infty}\frac{\lambda^k}{(k-1)!}              \\
       & =\lambda e^{-\lambda}\sum_{j=0}^{\infty}\frac{\lambda^j}{j!}=\lambda.
  \end{align*}
\end{example}

Till now, only countable sets and discrete random variables.

\subsection{Coupon Collector}

\begin{example}[Collecting a Complete Set]
  Collect $r$ types of coupons; receive a reward for a complete set.
  Assume all types equally likely and successive collections independent.

  Let $T=$ number of collections until a complete set is collected.
  Apparently $P(T<r)=0$.

  $P(T=n)$ for $n\in\{r,r+1,r+2,\ldots\}$?
  \[
    \underbrace{\{T=n\}\cup\{T>n\}}_{\text{disjoint}}=\{T>n-1\}
    \qquad\text{(immediately obvious)}.
  \]
  \begin{gather*}
    P(\{T=n\}\cup\{T>n\})=P(T=n)+P(T>n)=P(T>n-1),\\
    P(T=n)=P(T>n-1)-P(T>n).
  \end{gather*}
  $P(T>n)$: at least one coupon type missing.

  For $j\in\{1,2,\ldots,r\}$, define
  \[
    A_j=\{\text{haven't got type }j\text{ by the }n\text{th collection}\}.
  \]
  \[
    \{T>n\}=\bigcup_{j=1}^r A_j.
  \]
  \begin{align*}
    P\left(\bigcup_{j=1}^r A_j\right)
     & =\sum_{j=1}^r P(A_j)-\sum_{i<j}P(A_iA_j) \\
     & \quad+\sum_{i<j<k}P(A_iA_jA_k)-\cdots
    +(-1)^{r+1}P(A_1\cdots A_r).
  \end{align*}
  \[
    P(A_j)=\left(\frac{r-1}{r}\right)^n,\qquad
    P(A_iA_j)=\left(\frac{r-2}{r}\right)^n\quad(i\ne j).
  \]
  \begin{align*}
    P\left(\bigcup_{j=1}^r A_j\right)
     & =\binom r1\left(\frac{r-1}{r}\right)^n
    -\binom r2\left(\frac{r-2}{r}\right)^n                \\
     & \quad+\binom r3\left(\frac{r-3}{r}\right)^n-\cdots
    +(-1)^r\binom r{r-1}\left(\frac1r\right)^n            \\
     & =\sum_{\ell=1}^{r-1}(-1)^{\ell+1}\binom r\ell
    \left(\frac{r-\ell}{r}\right)^n.
  \end{align*}
\end{example}

\subsection{Variance}

\begin{definition}{Variance}{variance}
  If $X$ is a random variable with finite mean, its variance is
  \[
    \operatorname{Var}(X)=E[(X-EX)^2].
  \]
\end{definition}

Note: for $p(x)=P(X=x)$,
\[
  E[X]=\sum_x xp(x),\qquad E[f(X)]=\sum_x f(x)p(x).
\]
For finite second moment,
\begin{align*}
  \operatorname{Var}(X) & =E[(X-EX)^2]                                      \\
                        & =\sum_x(x-EX)^2p(x)                               \\
                        & =\sum_x\bigl(x^2-2EX\cdot x+(EX)^2\bigr)p(x)      \\
                        & =\sum_x x^2p(x)-2EX\sum_x xp(x)+(EX)^2\sum_x p(x) \\
                        & =E[X^2]-2E[X]^2+E[X]^2                            \\
                        & =E[X^2]-E[X]^2.
\end{align*}

\begin{example}[Bernoulli Variance]
  Parameter $p$: probability.
  \begin{gather*}
    p(0)=P(X=0)=1-p,\qquad p(1)=P(X=1)=p,\\
    E[X]=0\cdot(1-p)+1\cdot p=p,\\
    E[X^2]=0^2\cdot(1-p)+1^2\cdot p=p,\\
    \operatorname{Var}(X)=E[X^2]-E[X]^2=p-p^2.
  \end{gather*}
  \begin{center}
    \begin{tikzpicture}[x=4cm,y=5cm]
      \draw[->] (-.08,0)--(1.15,0) node[right] {$p$};
      \draw[->] (0,-.02)--(0,.34) node[above] {$\operatorname{Var}(X)$};
      \draw[blue,thick,domain=0:1,samples=60] plot (\x,{\x*(1-\x)});
      \draw (.5,.012)--(.5,-.012) node[below] {$\frac12$};
      \draw (1,.012)--(1,-.012) node[below] {$1$};
      \fill (.5,.25) circle[radius=1.5pt];
    \end{tikzpicture}
  \end{center}
\end{example}

\begin{example}[Discrete Uniform Variance]
  Uniform distribution on $\{1,2,\ldots,n\}$:
  \[
    P(X=j)=\frac1n\qquad\text{for }j=1,2,\ldots,n.
  \]
  \begin{align*}
    E[X]                  & =\sum_{j=1}^n j\cdot\frac1n
    =\frac1n\frac{n(n+1)}2=\frac{n+1}2,                                 \\
    E[X^2]                & =\sum_{j=1}^n j^2\cdot\frac1n
    =\frac1n\frac{n(n+1)(2n+1)}6=\frac{(n+1)(2n+1)}6,                   \\
    \operatorname{Var}(X) & =E[(X-EX)^2]=E[X^2]-E[X]^2                  \\
                          & =(n+1)\left(\frac{2n+1}6-\frac{n+1}4\right)
    =\frac{(n+1)(n-1)}{12}=\frac{n^2-1}{12}.
  \end{align*}
\end{example}

\begin{example}[Poisson Variance]
  Parameter $\lambda$: expected count in a time interval.
  \[
    p(n)=P(X=n)=\frac{\lambda^n}{n!}e^{-\lambda}.
  \]
  \begin{gather*}
    E[X]=\sum_{j=0}^{\infty}j\frac{\lambda^j}{j!}e^{-\lambda}
    =e^{-\lambda}\lambda e^\lambda=\lambda,\\
    E[X^2]=\sum_{j=0}^{\infty}j^2\frac{\lambda^j}{j!}e^{-\lambda}
    \qquad\text{(hard?)}
  \end{gather*}
  \begin{align*}
    E[X(X-1)] & =\sum_{j=0}^{\infty}j(j-1)\frac{\lambda^j}{j!}e^{-\lambda}    \\
              & =e^{-\lambda}\sum_{j=2}^{\infty}\frac{\lambda^j}{(j-2)!}      \\
              & =\lambda^2e^{-\lambda}\sum_{m=0}^{\infty}\frac{\lambda^m}{m!} \\
              & =\lambda^2=E[X^2]-E[X].
  \end{align*}
  \[
    E[X^2]=\lambda^2+\lambda,\qquad
    \operatorname{Var}(X)=E[X^2]-E[X]^2=\lambda.
  \]
\end{example}

\begin{example}[Exponential Function of a Poisson Random Variable]
  Suppose $X$ is a Poisson random variable with parameter $\lambda$ and
  $f(x)=e^{tx}$ for $t$ constant.
  \begin{align*}
    E[e^{tX}] & =\sum_{k=0}^{\infty}e^{tk}\frac{\lambda^k}{k!}e^{-\lambda} \\
              & =e^{-\lambda}\sum_{k=0}^{\infty}\frac{(\lambda e^t)^k}{k!} \\
              & =e^{-\lambda}e^{\lambda e^t}=e^{\lambda(e^t-1)}.
  \end{align*}
\end{example}

\subsection{Negative Binomial Distribution}

\begin{definition}{Negative Binomial Distribution}{negative-binomial-distribution}
  Parameters $r\in\{1,2,\ldots\}$ and $p\in(0,1)$.
  $X$ is the trial of the $r$th success in independent trials with success probability $p$.
  For $X\sim\operatorname{NBi}(r,p)$,
  \[
    P(X=n)=\binom{n-1}{r-1}p^r(1-p)^{n-r},\qquad n=r,r+1,\ldots.
  \]
\end{definition}

\[
  P(X=n)=\underbrace{\binom{n-1}{r-1}p^{r-1}(1-p)^{(n-1)-(r-1)}}_{r-1\text{ successes in the first }n-1\text{ trials}}\,p.
\]
\[
  \underbrace{\underset{1}{\rule{.5cm}{.4pt}}\quad
    \underset{2}{\rule{.5cm}{.4pt}}\quad\cdots\quad
    \underset{n-1}{\rule{.5cm}{.4pt}}}_{\text{first }n-1\text{ trials}}
  \quad\underset{n}{\text{success}}
\]
\begin{gather*}
  E[X]=\frac rp,\\
  \operatorname{Var}(X)=E[(X-EX)^2]=E[X^2]-E[X]^2
  =\frac rp\left(\frac{r+1}p-1\right)-\left(\frac rp\right)^2
  =\frac{r(1-p)}{p^2}.
\end{gather*}

\subsection{Hypergeometric Distribution}

\begin{definition}{Hypergeometric Distribution}{hypergeometric-distribution}
  An urn with $m$ white and $N-m$ black balls. Select $n$ balls at random
  without replacement. $X$: number of white balls selected.
  \[
    P(X=k)=\frac{\binom mk\binom{N-m}{n-k}}{\binom Nn},
    \qquad k=0,\ldots,m,
  \]
  with $\binom ab=0$ when $b<0$ or $b>a$.
\end{definition}

\begin{example}[Coyotes on Campus]
  Estimate coyotes on campus: $N$.\\
  Catch and release $m$ coyotes; tag them.\\
  Later catch $n$ coyotes. Suppose $c$ of these are tagged.

  Let $X=\#$ tagged and caught.
  \[
    P(X=i)=\frac{\binom mi\binom{N-m}{n-i}}{\binom Nn}.
  \]
  \[
    \operatorname*{arg\,max}_{N\ge m+n-c}
    \left\{\frac{\binom{N-m}{n-c}}{\binom Nn}\right\}.
  \]
\end{example}

\subsubsection{Hypergeometric Moments}

For $m,n\ge1$ and integer $k\ge1$,
\begin{align*}
  E[X^k] & =\sum_{i=0}^m i^kP(X=i)                                       \\
         & =\sum_{i=0}^m i^k\frac{\binom mi\binom{N-m}{n-i}}{\binom Nn}.
\end{align*}
By
\begin{align*}
  i\binom mi & =i\frac{m!}{i!(m-i)!}
  =m\frac{(m-1)!}{(i-1)!((m-1)-(i-1))!}
  =m\binom{m-1}{i-1},                           \\
  n\binom Nn & =\frac{N!}{(n-1)!((N-1)-(n-1))!}
  =N\binom{N-1}{n-1},
\end{align*}
\begin{align*}
  E[X^k] & =\sum_{i=1}^m i^{k-1}
  \frac{mn\binom{m-1}{i-1}\binom{N-m}{n-i}}{n\binom Nn}             \\
         & =mn\sum_{i=1}^m i^{k-1}
  \frac{\binom{m-1}{i-1}\binom{N-m}{n-i}}{N\binom{N-1}{n-1}}        \\
         & =\frac{mn}N\sum_{i=1}^m i^{k-1}
  \frac{\binom{m-1}{i-1}\binom{N-m}{(n-1)-(i-1)}}{\binom{N-1}{n-1}} \\
         & =\frac{mn}N\sum_{j=0}^{m-1}(j+1)^{k-1}
  \frac{\binom{m-1}j\binom{N-m}{n-1-j}}{\binom{N-1}{n-1}},\qquad j=i-1.
\end{align*}
The fraction is hypergeometric with parameters $N-1,m-1,n-1$.
Let
\[
  P(Z=j)=\frac{\binom{m-1}j\binom{N-m}{n-1-j}}{\binom{N-1}{n-1}}.
\]
Then
\[
  E[X^k]=\frac{mn}N E[(Z+1)^{k-1}].
\]
Consider:
\[
  E[X]=\frac{mn}N=np\qquad\text{(same as binomial)},\qquad p=\frac mN.
\]
For $N>1$,
\begin{align*}
  E[X^2]                & =\frac{mn}N E[Z+1]                                \\
                        & =\frac{mn}N\left(\frac{(m-1)(n-1)}{N-1}+1\right), \\
  \operatorname{Var}(X) & =E[X^2]-E[X]^2
  =np(1-p)\left(1-\frac{n-1}{N-1}\right).
\end{align*}
