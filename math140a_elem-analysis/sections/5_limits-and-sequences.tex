\subsection{Sequences}

\begin{definition}{Sequence}{sequence}
  A \textbf{sequence} is a function $S:\N\to\R$.
  \begin{flushleft}
    Notation: $S(n)$ or $S_n$\\
    or $(S_n)_{n=1}^{\infty}$ or $(S_n)_{n\in\N}$.
  \end{flushleft}
\end{definition}

\begin{example}
  \begin{gather*}
    S=\left(\frac1n\right)_{n\in\N}
    =\left(1,\frac12,\frac13,\ldots\right),\\
    S=\bigl((-1)^n\bigr)_{n\in\N}=(-1,1,-1,1,\ldots).
  \end{gather*}
\end{example}

\begin{note}
  Notation: $(\,)$ sequences, $\{\,\}$ sets.\\
  $(1,-1)$ is not $\{1,-1\}$.
\end{note}

\subsection{Convergence}

\begin{definition}{Convergence}{sequence-convergence}
  \textbf{Important!}
  \begin{flushleft}
    Let $S:\N\to\R$ be a sequence,\\
    let $t\in\R$.\\[0.5em]
    $S$ \textbf{converges to $t$}, or $S\to t$,
    or $\lim_{n\to\infty}S_n=t$,
  \end{flushleft}
  means
  \[
    \begin{gathered}
      \forall\varepsilon\in\R,\ \varepsilon>0,\\[0.5em]
      \exists N\in\N\text{ s.t.}\\[0.5em]
      n>N\implies\abs{S_n-t}<\varepsilon.
    \end{gathered}
  \]
\end{definition}

\begin{thinking}
  \begin{flushleft}
    $S_n\to t\iff$\\[0.5em]
    for every $\varepsilon>0$ (small window) challenging our assertion,\\[0.5em]
    we can find a $N\in\N$ (concrete index)\\[0.5em]
    such that for every $n>N$, $S_n$
    has sequence output: $\abs{S_n-t}<\varepsilon$.\\[0.5em]
    Distance $S_n\leftrightarrow t$
    is smaller than $\varepsilon$ window.
  \end{flushleft}
  \begin{center}
    \begin{tikzpicture}
      \draw[number line] (-3,0) -- (3,0);
      \draw[sketch accent] (-1.8,0) -- (1.8,0);
      \foreach \x/\label in {-1.8/{t-\varepsilon},1.8/{t+\varepsilon}} {
        \draw[orange!75!black,thick,fill=orange!5] (\x,0) circle (2.5pt);
        \node[above=6pt] at (\x,0) {$\label$};
      }
      \draw (0,-0.1) -- (0,0.1) node[below=5pt] {$t$};
      \node[sketch point] at (0.8,0) {};
      \node[below] (sn) at (1.4,-0.65) {$S_n$};
      \draw[->,>=latex] (sn) -- (0.8,-0.1);
    \end{tikzpicture}
  \end{center}
\end{thinking}

  $S_n\to0$: looks like so on the graph.

  \begin{thinking}[Sketch (intuition)]
    WTS
    (Definition~\ref{def:sequence-convergence}):
    \[
      \begin{gathered}
        \forall\varepsilon>0,\quad\exists N\in\N,\\[0.5em]
        \forall n>N,\quad\left|\frac1n-0\right|<\varepsilon.

\begin{example}[$1/n^2\to0$]
  Prove $\displaystyle\lim_{n\to\infty}\frac1{n^2}=0$.

  \begin{thinking}[Scratch work]
    WTS (Definition~\ref{def:sequence-convergence}):
    \[
      \begin{gathered}
        \forall\varepsilon>0,\quad\exists N\in\N\\
        \text{s.t. }n>N\implies\left|\frac1{n^2}-0\right|<\varepsilon.
      \end{gathered}
    \]
    Given $\varepsilon>0$, how to choose $N$?
    \begin{align*}
      \text{We need } & \frac1{n^2}<\varepsilon,    \\
                      & \frac1\varepsilon<n^2,      \\
                      & \frac1{\sqrt\varepsilon}<n.
    \end{align*}
    Need $N>\dfrac1{\sqrt\varepsilon}$.
  \end{thinking}

  \begin{proof}
    For arbitrary $\varepsilon>0$:\\
    Choose $N\in\N$, s.t.\ $N>\dfrac1{\sqrt\varepsilon}$.\\
    If $n>N$, $n>\dfrac1{\sqrt\varepsilon}$, hence $\dfrac1{n^2}<\varepsilon$.
    \[
      \implies\left|\frac1{n^2}\right|<\varepsilon.\qedhere
    \]
  \end{proof}
\end{example}

\subsection{Uniqueness of Limits}

\begin{theorem}{Uniqueness of Sequence Limits}{unique-sequence-limit}
  For a sequence $(S_n)$,
  \[
    S_n\to t_1\text{ and }S_n\to t_2\implies t_1=t_2.
  \]
\end{theorem}

\begin{lemma}{Arbitrarily Small Distance}{arbitrarily-small-distance}
  For $a,b\in\R$,
  \[
    \bigl(\abs{a-b}<\varepsilon\quad\forall\varepsilon>0\bigr)
    \iff a=b.
  \]
\end{lemma}

\begin{proof}
  $(\Leftarrow)$ Trivial.

  $(\Rightarrow)$ Contrapositive:
  \[
    a\ne b\implies\exists\varepsilon>0,\quad\abs{a-b}\ge\varepsilon.\qedhere
  \]
\end{proof}

\newpage
\begin{proof}[Proof of Theorem~\ref{thm:unique-sequence-limit}]
  By assumption:
  \begin{enumerate}
    \item $\forall\varepsilon_1>0,\quad\exists N_1\in\N$, s.t.
          \[
            n>N_1\implies\abs{S_n-t_1}<\varepsilon_1.
          \]
    \item $\forall\varepsilon_2>0,\quad\exists N_2\in\N$, s.t.
          \[
            n>N_2\implies\abs{S_n-t_2}<\varepsilon_2.
          \]
  \end{enumerate}
  WTS, by Lemma~\ref{lem:arbitrarily-small-distance},
  \[
    t_1=t_2\iff\forall\varepsilon>0,\quad\abs{t_1-t_2}<\varepsilon.
  \]

  \begin{thinking}[Trick (observation)]
    Note that
    \begin{align*}
      \abs{t_1-t_2}
       & \le\abs{(S_n-t_1)-(S_n-t_2)}                  \\
       & \le\underbrace{\abs{S_n-t_1}+\abs{S_n-t_2}}_{
        \text{can get arbitrarily small by assumption}}.
    \end{align*}
    \begin{center}
      \begin{tikzpicture}
        \node[sketch point,label=below:{$t_1$}] (t1) at (0,0) {};
        \node[sketch point,label=above:{$S_n$}] (sn) at (1,1) {};
        \node[sketch point,label=below:{$t_2$}] (t2) at (2,0) {};
        \draw[->,>=latex,sketch accent] (0.15,0) -- (1.85,0);
        \draw[->,>=latex,sketch accent] (0.15,0.15)
          .. controls (0.5,1.28) and (1.5,1.28) .. (1.85,0.15);
      \end{tikzpicture}
    \end{center}
  \end{thinking}

  Let $\varepsilon>0$.\\
  By assumption, choose
  \begin{align*}
    N_1\in\N, & \quad n>N_1\implies\abs{S_n-t_1}<\frac\varepsilon2, \\
    N_2\in\N, & \quad n>N_2\implies\abs{S_n-t_2}<\frac\varepsilon2.
  \end{align*}
  Choose $n\in\N$ s.t.\ $n>\max\{N_1,N_2\}$.
  \begin{align*}
    \text{Since }\abs{t_1-t_2}
     & \le\abs{S_n-t_1}+\abs{S_n-t_2}                    \\
     & <\frac\varepsilon2+\frac\varepsilon2=\varepsilon.
  \end{align*}
  Proven.
\end{proof}

\subsection{Further Limit Examples}

\begin{example}
  \[
    \lim_{n\to\infty}\frac{5n+1}{7n-3}=\frac57.
  \]
  \begin{thinking}[Proof sketch]
    WTS:
    \[
      \forall\varepsilon>0,\quad\exists N\in\N,\quad
      n>N\implies\left|\frac{5n+1}{7n-3}-\frac57\right|<\varepsilon.
    \]
    \begin{align*}
      \left|\frac{5n+1}{7n-3}-\frac57\right|
       & =\left|\frac{7(5n+1)-5(7n-3)}{7(7n-3)}\right|      \\
       & =\left|\frac{22}{49n-21}\right|=\frac{22}{49n-21}.
    \end{align*}
    Find $n$ big enough s.t.
    \begin{align*}
      \frac{22}{49n-21}                      & <\varepsilon, \\
      \frac{22}{\varepsilon}                 & <49n-21,      \\
      \frac{22}{49\varepsilon}+\frac{21}{49} & <n.
    \end{align*}
  \end{thinking}
  \begin{proof}
    Let $\varepsilon>0$. Choose $N\in\N$ s.t.
    \[
      N>\frac{22}{49\varepsilon}+\frac{21}{49}.
    \]
    For $n>N$,
    \[
      \frac{22}{49\varepsilon}+\frac{21}{49}<n
      \implies\frac{22}{49n-21}<\varepsilon.
    \]
    Also,
    \[
      \left|\frac{5n+1}{7n-3}-\frac57\right|
      =\left|\frac{22}{49n-21}\right|
      =\frac{22}{49n-21}<\varepsilon.\qedhere
    \]
  \end{proof}
\end{example}

\begin{example}[A divergent sequence]
  $S_n=(-1)^n$, so $(S_n)=(-1,1,-1,1,\ldots)$, does not converge.

  \begin{proof}
    WTS: $\forall t\in\R,\ \neg(S_n\to t)$.
    Fix $t\in\R$. WTS:
    \begin{align*}
             & \neg\bigl(\forall\varepsilon>0,\ \exists N\in\N,\
      \forall n>N,\ \abs{S_n-t}<\varepsilon\bigr)                            \\
      \iff{} & \exists\varepsilon>0,\
      \neg\bigl(\exists N\in\N,\ \forall n>N,\ \abs{S_n-t}<\varepsilon\bigr) \\
      \iff{} & \exists\varepsilon>0,\ \forall N\in\N,\
      \neg\bigl(\forall n>N,\ \abs{S_n-t}<\varepsilon\bigr)                  \\
      \iff{} & \exists\varepsilon>0,\ \forall N\in\N,\
      \exists n>N,\ \abs{S_n-t}\ge\varepsilon.
    \end{align*}
    Let $\varepsilon=\frac12$. WTS:
    \[
      \forall N\in\N,\quad\exists n>N,\quad\abs{S_n-t}\ge\frac12.
    \]
    For $\forall N\in\N$:
    \begin{enumerate}
      \item Assume $N$ even. So $S_{N+1}=-1$.\\
            WTS $\abs{-1-t}\ge\frac12$.\\
            Suppose instead $\abs{-1-t}<\frac12$.
            \begin{thinking}
              \begin{center}
                \begin{tikzpicture}[x=1.8cm]
                  \draw[number line] (-2,0) -- (1.4,0);
                  \draw[sketch accent] (-1.5,0) -- (-0.5,0);
                  \foreach \x/\lab in {-1.5/{-\frac32},-0.5/{-\frac12}} {
                    \draw[orange!75!black,thick,fill=orange!5] (\x,0) circle (2.5pt);
                    \node[below=6pt] at (\x,0) {$\lab$};
                  }
                  \foreach \x in {-1,1} {
                    \draw (\x,-0.1) -- (\x,0.1) node[below=5pt] {$\x$};
                  }
                  \node[sketch point] at (-0.8,0) {};
                  \node (t) at (-0.35,0.65) {$t$};
                  \draw[->,>=latex] (t) to[bend right] (-0.8,0.08);
                \end{tikzpicture}
              \end{center}
              If $t$ inside, step out of window.
            \end{thinking}
            WTS: $\abs{S_{N+2}-t}\ge\frac12$.
            \[
              2=\abs{(1-t)-(-1-t)}
              \le\abs{1-t}+\abs{-1-t}<\abs{1-t}+\frac12.
            \]
            Thus $\abs{S_{N+2}-t}=\abs{1-t}>\frac32>\frac12$.
      \item $N$ odd: vice versa.
    \end{enumerate}
  \end{proof}
\end{example}

\begin{theorem}{Limit of Square Roots}{limit-square-roots}
  Let $(S_n)$ be a sequence s.t.\ $\forall n,\ S_n>0$ and $S_n\to t$.
  Then
  \[
    \lim_{n\to\infty}\sqrt{S_n}=\sqrt{t}.
  \]
\end{theorem}

\begin{thinking}[Proof sketch]
  WTS:
  \[
    \forall\varepsilon>0,\quad\exists N\in\N\text{ s.t. }
    n>N\implies\abs{\sqrt{S_n}-\sqrt{t}}<\varepsilon.
  \]
  \begin{flushleft}
    We know $\abs{S_n-t}<\varepsilon$ for $n$ large.\\
    Connect the two?
  \end{flushleft}
\end{thinking}
\begin{thinking}[Trick]
  \[
    S_n-t=(\sqrt{S_n}-\sqrt{t})(\sqrt{S_n}+\sqrt{t}).
  \]
  So
  \[
    \sqrt{S_n}-\sqrt{t}
    =\frac{\overbrace{S_n-t}^{\text{what we know}}}
    {\underbrace{\sqrt{S_n}+\sqrt{t}}_{\text{some control?}}}.
  \]
  If $t>0$,
  \[
    \abs{\sqrt{S_n}-\sqrt{t}}
    =\frac{\abs{S_n-t}}{\abs{\sqrt{S_n}+\sqrt{t}}}
    <\frac{\varepsilon}{\sqrt{S_n}+\sqrt{t}}
    \le\frac{\varepsilon}{\sqrt{t}}
    \qquad(\sqrt{S_n}\text{ nonnegative}).
  \]
  \begin{flushleft}
    If $t=0$: WTS $\abs{\sqrt{S_n}}<\varepsilon$ for large $n$.\\
    If $\abs{S_n}<\varepsilon^2$, then $\abs{\sqrt{S_n}}=\sqrt{S_n}<\varepsilon$.\\
    If $t<0$?
  \end{flushleft}
\end{thinking}

\begin{lemma}{Lower Bounds Pass to the Limit}{limit-lower-bound}
  If $\lim_{n\to\infty}S_n=t$ and $\exists r\in\R$ s.t.\ $S_n\ge r\ \forall n$,
  then $t\ge r$.
\end{lemma}

If every term is bounded below, the limit should not be smaller.

\begin{proof}[Formal proof of Theorem~\ref{thm:limit-square-roots}]
  By Lemma~\ref{lem:limit-lower-bound}, $t\ge0$.\\
  Two cases: $t=0$ and $t>0$.
  \begin{enumerate}
    \item $t=0$. Let $\varepsilon$ be arbitrary $>0$.\\
          Choose $N\in\N$ s.t.\ $n>N\implies\abs{S_n}<\varepsilon^2$\\
          (we know this by definition of $S_n\to t$).\\
          For $n>N$, we have
          \[
            \abs{\sqrt{S_n}-0}=\sqrt{S_n}<\sqrt{\varepsilon^2}=\varepsilon.
          \]
    \item $t>0$. Let arbitrary $\varepsilon>0$. Choose $N\in\N$ s.t.
          \[
            n>N\implies\abs{S_n-t}<\sqrt{t}\,\varepsilon.
          \]
          For $n>N$,
          \begin{align*}
            \abs{\sqrt{S_n}-\sqrt{t}}
             & =\frac{\abs{S_n-t}}{\sqrt{S_n}+\sqrt{t}}              \\
             & <\frac{\sqrt{t}\,\varepsilon}{\sqrt{S_n}+\sqrt{t}}    \\
             & \le\frac{\sqrt{t}\,\varepsilon}{\sqrt{t}}=\varepsilon
            \qquad(\sqrt{S_n}\ge0).
          \end{align*}
  \end{enumerate}
  Both cases resolved.
\end{proof}

\begin{proof}

    t=\lim_{n\to\infty}S_n
    \implies\exists N\in\N\text{ s.t. }n>N\implies\abs{S_n-t}<1.

  \begin{thinking}[Why]
    \begin{center}

        \draw[number line] (-3,0) -- (3,0);
        \node at (-2,0) {$($};
        \node at (2,0) {$)$};

  \[
    \abs{S_n}\le M<2M.\qedhere

\begin{theorem}{Sequence Limit Laws}{sequence-limit-laws}
  $(S_n)$, $(t_n)$ converge.
  \begin{enumerate}[label=(\roman*)]
    \item $\displaystyle\lim_{n\to\infty}(S_n+t_n)
            =\lim_{n\to\infty}S_n+\lim_{n\to\infty}t_n$.
    \item $\displaystyle\forall k\in\R,\quad\lim_{n\to\infty}(kS_n)
            =k\lim_{n\to\infty}S_n$.
    \item $\displaystyle\lim_{n\to\infty}(S_nt_n)
            =\left(\lim_{n\to\infty}S_n\right)\cdot\left(\lim_{n\to\infty}t_n\right)$.
    \item $\displaystyle S_n\ne0\text{ and }\lim_{n\to\infty}S_n\ne0$
          \[
            \implies\lim_{n\to\infty}\frac{t_n}{S_n}
            =\frac{\lim_{n\to\infty}t_n}{\lim_{n\to\infty}S_n}.
          \]
  \end{enumerate}
\end{theorem}

\begin{proof}[Proof of Theorem~\ref{thm:sequence-limit-laws}(i)]
  Let
  \[
    \lim_{n\to\infty}S_n=p,\qquad\lim_{n\to\infty}t_n=q.
  \]
  Let $\varepsilon>0$.\\
  WTS $\abs{S_n+t_n-p-q}<\varepsilon$ for $n$ large.

  Observe
  \begin{align*}
    \abs{S_n+t_n-(p+q)}
     & =\abs{(S_n-p)+(t_n-q)}      \\
     & \le\abs{S_n-p}+\abs{t_n-q}.
  \end{align*}
  Choose $N\in\N$ s.t.
  \[
    n>N\implies
    \begin{cases}
      \abs{S_n-p}<\varepsilon/2, \\
      \abs{t_n-q}<\varepsilon/2.
    \end{cases}
  \]
  $N$ exists by taking the maximum of the two convergence thresholds.
  \[
    \abs{S_n-p}+\abs{t_n-q}<\frac\varepsilon2+\frac\varepsilon2
    =\varepsilon.\qedhere
  \]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:sequence-limit-laws}(iv)]
  Let $\lim_{n\to\infty}S_n=p$, $\lim_{n\to\infty}t_n=q$.
  Enough to show
  \[
    \lim_{n\to\infty}\frac1{S_n}=\frac1p
    \qquad\text{(by (iii)).}
  \]
  WTS: $\left|\dfrac1{S_n}-\dfrac1p\right|<\varepsilon$
  for $\forall\varepsilon>0$ and $n$ large.
  \[
    \left|\frac1{S_n}-\frac1p\right|
    =\left|\frac{S_n-p}{S_np}\right|
    =\frac{\overbrace{\abs{S_n-p}}^{\text{can bound}}}
    {\underbrace{\abs{S_np}}_{\text{???}}}.
  \]

  \begin{lemma}{Bounded Away from Zero}{sequence-away-from-zero}
    $\abs{S_n}\abs p$ bounded below by a positive number.\\
    Formally, for $S_n\ne0$ and $\lim S_n=p\ne0$,
    \[
      \inf\{\abs{S_n}\mid n\in\N\}>0.
    \]
  \end{lemma}
  \begin{thinking}
    Important in a lot of proofs.\\
    Let $S_n\to p\ne0$.
    \begin{center}
      \begin{tikzpicture}
        \draw[number line] (-0.5,0) -- (4.5,0);
        \draw (0,-0.1) -- (0,0.1) node[below=5pt] {$0$};
        \draw (2.5,-0.1) -- (2.5,0.1) node[above=5pt] {$\abs p$};
        \foreach \x in {1.5,3.5} {
          \draw[orange!75!black,thick,fill=orange!5] (\x,0) circle (2.5pt);
        }
        \draw[decorate,decoration={brace,mirror,amplitude=5pt}]
          (1.5,-0.25) -- (3.5,-0.25)
          node[midway,below=8pt,align=center] {all but finitely many\\$\abs{S_n}$ here};
      \end{tikzpicture}
    \end{center}
  \end{thinking}
  Let $m=\inf\{\abs{S_n}\mid n\in\N\}>0$.\\
  Can choose $N$ s.t.\ $n>N\implies\abs{S_n-p}<\varepsilon m\abs p$.
  \[
    \frac{\abs{S_n-p}}{\abs{S_n}\abs p}
    \le\frac{\abs{S_n-p}}{m\abs p}
    <\frac{\varepsilon m\abs p}{m\abs p}=\varepsilon.\qedhere
  \]
\end{proof}

\subsubsection{More Limit Theorems}

\begin{theorem}{Power and Root Limits}{power-root-limits}
  \begin{enumerate}[label=(\roman*)]
    \item $\displaystyle\lim_{n\to\infty}\frac1{n^p}=0$ for $\forall p>0$.
    \item $\displaystyle\lim_{n\to\infty}a^n=0$ if $\abs a<1$.
    \item $\displaystyle\lim_{n\to\infty}n^{1/n}=1$.
    \item $\displaystyle\lim_{n\to\infty}a^{1/n}=1$, $\forall a>0$.
  \end{enumerate}
\end{theorem}

\noindent\textit{Proof of (ii) (incomplete).}
\begin{enumerate}
  \item $\abs a=0$: trivial.
  \item $\abs a>0$: WTS $\abs{a^n}$ is small for $n$ large.\\
        Let $\varepsilon>0$.
        \begin{align*}
          \text{WTS }\abs{a^n}<\varepsilon
           & \iff\abs a^n<\varepsilon
          \iff\ln(\abs a^n)<\ln\varepsilon                    \\

           & \iff n>\frac{\ln\varepsilon}{\ln\abs a}.
        \end{align*}
        For $\varepsilon$, choose $N\in\N$ s.t.\ $N>\dfrac{\ln\varepsilon}{\ln\abs a}$.
        \[
          n>N\implies\abs{a^n}<\varepsilon.

\end{enumerate}

\end{proof}

\subsection{Monotone and Cauchy Sequences}

Goal: how to say $S_n$ converges without knowing the limit explicitly?
\begin{example}
  $\displaystyle S_n=\sum_{k=1}^n\frac{(-1)^k}{k}$: hard to calculate!
\end{example}

\subsubsection{Monotonicity}

\begin{definition}{Monotone Sequence}{monotone-sequence}
  $(S_n)$ a sequence.
  \begin{align*}
    (S_n)\text{ is increasing} & \iff\forall n\in\N,\quad S_n\le S_{n+1}, \\
    (S_n)\text{ is decreasing} & \iff\forall n\in\N,\quad S_n\ge S_{n+1}.
  \end{align*}
  $(S_n)$ is monotonic $\iff$ increasing OR decreasing.
\end{definition}

\begin{example}
  $S_n=1/n$: decreasing.\qquad $S_n=n^2$: increasing.\\
  $S_n=(-1)^n/n$: not monotonic.
\end{example}

\begin{theorem}{Monotone Convergence Theorem}{monotone-convergence}
  Sequence $(S_n)$ is bounded and monotonic $\implies(S_n)$ converges.
\end{theorem}

\begin{note}
  Converse is not true.
\end{note}
\begin{example}[Counterexample]
  $(-1)^n/n$ converges but is not monotonic.
\end{example}

\begin{note}
  Statement is false in $\Q$.
\end{note}
\begin{example}[Counterexample]
  $\sqrt2=1.41421\ldots$
  \begin{flushleft}
    $(S_n)=(1,1.4,1.41,1.414,1.4142,\ldots)$\\
    is bounded and monotonic\\
    but does not converge in $\Q$.
  \end{flushleft}
\end{example}
(We need to use completeness of $\R$!)

\begin{proof}[Proof of Theorem~\ref{thm:monotone-convergence}]
  Assume $S_n$ increasing.\\
  Let $S=\{S_n\mid n\in\N\}$.
  \[
    S_n\text{ bounded and in }\R\implies\exists\sup S.
  \]
  WTS $S_n\to\sup S$.\\
  WTS $N$, $n>N\implies\abs{S_n-\sup S}<\varepsilon$.

  Let $\varepsilon>0$.\\
  By definition of $\sup S$, $\sup S-\varepsilon$ is not an upper bound of $S$.
  \[
    \exists u\in S\text{ s.t. }\sup S-\varepsilon<u
    \qquad\text{(negation of upper bound).}
  \]
  Let $N$ s.t.\ $u=S_N$.\\
  For $n>N$, since $(S_n)$ increasing,
  \[
    \sup S-\varepsilon<S_N\le S_n.
  \]
  Also $S_n\le\sup S<\sup S+\varepsilon$.
  \begin{gather*}
    \sup S-\varepsilon<S_n<\sup S+\varepsilon,\\
    \abs{S_n-\sup S}<\varepsilon.
  \end{gather*}
  For $(S_n)$ decreasing, apply previous argument to $(-S_n)$.
\end{proof}

\subsubsection{Cauchy Sequences}

\begin{definition}{Cauchy Sequence}{cauchy-sequence}
  $(S_n)$ is Cauchy $\iff$
  \[
    \forall\varepsilon>0,\quad\exists N\in\N\text{ s.t. }
    n,m>N\implies\abs{S_n-S_m}<\varepsilon.
  \]
\end{definition}

\begin{thinking}[Intuition]
  As we go far in the sequence,\\
  the terms are ``bunching up'': $\abs{S_n-S_m}$ becomes small.
\end{thinking}

\begin{theorem}{Cauchy Criterion}{cauchy-criterion}
  $(S_n)$ converges $\iff(S_n)$ is Cauchy.
\end{theorem}

Upshot: to prove convergence, only need Cauchy.

\begin{proof}[$(\Rightarrow)$]
  For $S_n\to t$:\\
  WTS $\forall\varepsilon>0$, $\exists N\in\N$ s.t.\ $m,n>N\implies\abs{S_n-S_m}<\varepsilon$.

  Let $\varepsilon>0$ be arbitrary. Choose $N$ s.t.
  \[
    n>N\implies\abs{S_n-t}<\frac\varepsilon2.
  \]
  For $m,n>N$,
  \begin{align*}
    \abs{S_n-S_m} & =\abs{(S_n-t)-(S_m-t)}     \\
                  & \le\abs{S_n-t}+\abs{S_m-t}
    <\frac\varepsilon2+\frac\varepsilon2=\varepsilon.\qedhere
  \end{align*}
\end{proof}

$(\Leftarrow)$: hard!

\subsection{Limit Superior and Limit Inferior}

\begin{definition}{Limit Superior and Limit Inferior}{limsup-liminf}
  Auxiliary definition: $(S_n)$ a sequence.
  \begin{enumerate}[label=(\roman*)]
    \item Define $\displaystyle\limsup S_n:=\lim_{N\to\infty}\bigl(\sup\{S_n\mid n>N\}\bigr)$.
    \item Define $\displaystyle\liminf S_n:=\lim_{N\to\infty}\bigl(\inf\{S_n\mid n>N\}\bigr)$.
  \end{enumerate}
\end{definition}

\begin{thinking}[Intuition]
  Limit of $t_N=\sup\{S_n\mid n>N\}$:
  \[
    \limsup S_n=\lim t_N.
  \]
\end{thinking}

\begin{example}
  \[
    S_n=\frac{(-1)^n}{n}+(-1)^n
    \quad\text{or}\quad (-1)^n\left(1+\frac1n\right).
  \]
  \begin{center}
    \begin{tikzpicture}[x=0.85cm,y=1.05cm]
      \draw[->,>=latex] (-0.4,0) -- (6.8,0) node[right] {$n$};
      \draw[->,>=latex] (0,-2.35) -- (0,1.95) node[above] {$S_n$};
      \foreach \y in {-2,-1,1} {
        \draw (0.08,\y) -- (-0.08,\y) node[left] {$\y$};
        \draw[dashed,gray] (0.15,\y) -- (6.5,\y);
      }
      \foreach \n in {1,...,6} {
        \draw (\n,0.08) -- (\n,-0.08) node[below] {$\n$};
        \node[sketch point] at (\n,{(-1)^\n*(1+1/\n)}) {};
      }
    \end{tikzpicture}
  \end{center}
  \[
    \limsup S_n=1,\qquad\liminf S_n=-1.
  \]
\end{example}

\begin{example}
  $S_n=n$. $\limsup S_n$ does not exist.
\end{example}

\begin{theorem}{Existence of Limit Superior and Limit Inferior}{bounded-limsup-liminf}
  $(S_n)$ bounded $\implies\limsup S_n$ and $\liminf S_n$ exist.
\end{theorem}

\begin{proof}[Proof that $\limsup$ exists]
  Let $t_N=\sup\{S_n\mid n>N\}$. WTS $\exists\lim_{N\to\infty}t_N$.

  Use the monotone convergence theorem (Theorem~\ref{thm:monotone-convergence}):
  WTS $t_N$ bounded and monotone.
  \begin{gather*}
    S_n\text{ bounded}:\quad\exists M>0\text{ s.t. }\abs{S_n}<M.\\
    \text{So }t_N=\sup\{S_n\mid n>N\}\le M.\\
    \text{Similarly, }t_N\ge-M.\\
    \implies\abs{t_N}\le M.
  \end{gather*}
  $t_N$ monotone (decreasing), because
  \[
    \{S_n\mid n>N+1\}\subseteq\{S_n\mid n>N\}.
  \]
  \[
    t_{N+1}=\sup\{S_n\mid n>N+1\}
    \le\sup\{S_n\mid n>N\}=t_N.
  \]
  Note: $A\subseteq B\implies\sup A\le\sup B$.
\end{proof}

\begin{theorem}{Convergence and Limit Superior/Inferior}{convergence-limsup-liminf}
  For a sequence:
  \begin{gather*}
    \exists\lim\implies\lim=\limsup=\liminf.\\
    \limsup=\liminf\implies\exists\lim
    \land\lim=\limsup=\liminf.
  \end{gather*}
\end{theorem}

\noindent WTS for Theorem~\ref{thm:convergence-limsup-liminf}:
\begin{enumerate}[label=(\roman*)]
  \item Assume $\lim S_n=p$. ETS $\limsup S_n\le p$, $\liminf S_n\ge p$.
  \item $\limsup S_n=\liminf S_n\implies\exists\lim S_n$,
        and $\limsup S_n=\liminf S_n=\lim S_n$.
\end{enumerate}

\begin{proof}[Proof of (i)]
  To show $\limsup S_n\le p$, let $\varepsilon>0$. WTS
  \[
    \limsup S_n\le p+\varepsilon.
  \]
  Since $S_n\to p$, $\exists M\in\N$ s.t.
  \[
    n>M\implies\abs{S_n-p}<\varepsilon.
  \]
  (Note $n>M\implies S_n<p+\varepsilon$.)\\
  Note that $p+\varepsilon$ is an upper bound for $\{S_n\mid n>M\}$.
  \[
    \implies\underbrace{t_M}_{\text{least upper bound}}
    \le\underbrace{p+\varepsilon}_{\text{upper bound}}.
  \]
  Key: $(t_N)$ decreasing $\implies t_N\le p+\varepsilon$ for $\forall N\ge M$.
  \[
    \text{By now: }\lim t_N\le p+\varepsilon
    \iff\limsup S_n\le p+\varepsilon.
  \]
  Showing $\liminf S_n\ge p$: similar.
\end{proof}

\begin{proof}[Proof of (ii)]
  \[
    \limsup S_n=\liminf S_n
    \implies\limsup S_n=\liminf S_n=\lim S_n.
  \]
  Let
  \begin{align*}
    a_N & =\inf\{S_n\mid n\ge N\}, \\
    b_N & =\sup\{S_n\mid n\ge N\}.
  \end{align*}
  \[
    \implies a_N\le S_N\le b_N,\qquad\forall N\in\N.
  \]
  But
  \[
    \lim a_N=\liminf S_n=\limsup S_n=\lim b_N.
  \]
  By squeeze theorem,
  \[
    \lim S_N=\lim a_N=\lim b_N.\qedhere
  \]
\end{proof}

\begin{theorem}{Cauchy Sequences Are Bounded}{cauchy-sequence-bounded}
  $S_n$ Cauchy $\implies S_n$ bounded.
\end{theorem}

\begin{theorem}{Cauchy Sequences Converge}{cauchy-sequence-converges}
  Cauchy $\implies$ convergence.
\end{theorem}

\noindent Sketch: show $\limsup=\liminf$.

\begin{proof}
  $S_n$ Cauchy $\implies S_n$ bounded
  $\implies\exists\limsup S_n,\liminf S_n$.

  Since $\limsup S_n\ge\liminf S_n$,\\
  enough to show $\limsup S_n\le\liminf S_n$\\
  (to make $\limsup S_n=\liminf S_n$).
  \[
    \text{ETS: }\forall\varepsilon>0,\quad
    \limsup S_n\le\liminf S_n+\varepsilon.
  \]
  Let $\varepsilon>0$.
  \begin{align*}
    a_N & =\sup\{S_n\mid n>N\}, \\
    b_N & =\inf\{S_n\mid n>N\}.
  \end{align*}
  \[
    \implies\limsup S_n=\lim a_N,\qquad
    \liminf S_n=\lim b_N
    \quad\text{(by Definition~\ref{def:limsup-liminf}).}
  \]
  \[
    (S_n)\text{ Cauchy}\implies\exists N_0,\quad
    m,n>N_0\implies\abs{S_n-S_m}<\varepsilon.
  \]
  In particular, $n,m>N_0\implies S_n<S_m+\varepsilon$.\\
  Or in other words,
  \[
    m>N_0\implies S_m+\varepsilon\text{ is an upper bound for }
    \{S_n\mid n>N_0\}.
  \]
  \[
    a_{N_0}=\sup\{S_n\mid n>N_0\}\le S_m+\varepsilon,
    \qquad\forall m>N_0.
  \]
  \begin{align*}
    \implies a_{N_0}-\varepsilon\le S_m
     & \iff a_{N_0}-\varepsilon\text{ is a lower bound for }
    \{S_m\mid m>N_0\},                                       \\
    \implies a_{N_0}-\varepsilon
     & \le\inf\{S_m\mid m>N_0\}=b_{N_0}.
  \end{align*}
  Since $(a_N)$ decreasing, $(b_N)$ increasing,
  \[
    \implies\limsup S_n\le a_{N_0}\le b_{N_0}+\varepsilon
    \le\liminf S_n+\varepsilon.\qedhere
  \]
\end{proof}

\subsection{Subsequences}

\begin{definition}{Subsequence}{subsequence}
  Let $(S_n)_{n\in\N}$ be a sequence.

  A subsequence is $(t_k)_{k\in\N}$ s.t.\ $\forall k\in\N$,
  $\exists n_k\in\N$ s.t.
  \begin{enumerate}[label=(\roman*)]
    \item $t_k=S_{n_k}$,
    \item $\forall k,\quad n_k<n_{k+1}$.
  \end{enumerate}
\end{definition}

\begin{thinking}[Intuition]
  $(t_k)$ has terms of $(S_n)$ in the same order.\\
  $n$ is a remap of indices: $\forall k\in\N$ in $t$, $k\mapsto n_k$ for $S$.
\end{thinking}

\begin{example}
  $S_n=\dfrac1n$; $t_k=\dfrac1{2k}$ with $n_k=2k$ is a subsequence of $(S_n)$.

Next big goal: $\lim(S_n)$ and $\lim(t_k)$?

\begin{theorem}{Limit of a Subsequence}{subsequence-limit}
  $(t_k)$ a subsequence of $(S_n)$.
  \[
    (S_n)\text{ converges}\implies(t_k)\text{ also converges}
    \quad\text{and}\quad\lim S_n=\lim t_k.
  \]
\end{theorem}

\begin{proof}
  Suppose $\lim S_n=r$.\\
  Let $\varepsilon>0$.
  \[
    \exists N\text{ s.t. }n>N\implies\abs{S_n-r}<\varepsilon.
  \]
  Suppose $k>N$. We know $\exists n_k$ s.t.\ $t_k=S_{n_k}$,
  where $n_k\ge k>N$.
  \[
    \abs{t_k-r}=\abs{S_{n_k}-r}<\varepsilon.\qedhere
  \]
\end{proof}

What if $(S_n)$ does not converge?

\begin{example}
  \begin{enumerate}
    \item For $S_n=n$, $(S_n)$ has no convergent subsequences.
    \item For $S_n=(-1)^n\left(1+\dfrac1n\right)$,
          \begin{align*}
            t_k & =S_{2k}=1+\frac1{2k}\to1,                    \\
            r_k & =S_{2k+1}=-\left(1+\frac1{2k+1}\right)\to-1.
          \end{align*}
  \end{enumerate}
\end{example}

\begin{theorem}{Bolzano--Weierstrass Theorem}{bolzano-weierstrass}
  $(S_n)$ bounded $\implies\exists$ convergent subsequence.
\end{theorem}

\begin{lemma}{Monotone Subsequence}{monotone-subsequence}

\begin{theorem}{Limit Superior and Inferior as Subsequential Limits}{limsup-liminf-subsequences}
  $(S_n)$ bounded $\implies\exists$ subsequences $(t_k)$, $(r_k)$ s.t.
  \[
    t_k\to\limsup S_n,\qquad r_k\to\liminf S_n.
  \]
\end{theorem}

\begin{proof}[Proof of the $\limsup$ subsequence]
  Note: we know $\forall$ sequences, there is a monotone subsequence.

  By Theorem~\ref{thm:subsequential-limit-criterion}, ETS
  \[
    \forall\varepsilon>0,\quad
    \{n\in\N\mid\abs{S_n-\limsup S_n}<\varepsilon\}\text{ is infinite}.
  \]
  For $\forall\varepsilon>0$, since
  \[
    \limsup S_n=\lim_{N\to\infty}\sup\{S_n\mid n>N\},
    \qquad t_N=\sup\{S_n\mid n>N\},
  \]
  \[
    \exists N_0\in\N\text{ s.t. }N\ge N_0
    \implies\abs{t_N-\limsup S_n}<\varepsilon.
  \]
  \[
    S_n<\limsup S_n+\varepsilon,\qquad\forall n>N_0.
  \]
  Suppose for contradiction that
  $\{n\in\N\mid\abs{S_n-\limsup S_n}<\varepsilon\}$ is finite.

  It must be that $\exists N_1\ge N_0$ s.t.
  \[
    S_n\le\limsup S_n-\varepsilon,\qquad\forall n>N_1.
  \]

  \begin{thinking}
    \begin{center}
      \begin{tikzpicture}[x=1cm,y=1cm,>=latex]

      \node at (-1.8,0) {\large$( $};

      \node[above left] at (-1.8,0.25) {$\limsup S_n-\varepsilon$};
      \node[above] at (0,0.25) {$\limsup S_n$};
      \node[above right] at (1.8,0.25) {$\limsup S_n+\varepsilon$};

  Since $n>N_1\implies S_n\le\limsup S_n-\varepsilon$,
  \[
    \implies\limsup S_n\le\limsup S_n-\varepsilon.
  \]
  Contradiction.
\end{proof}
