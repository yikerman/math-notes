\subsection{Differential Equations and Initial Value Problems}

Differential equations are used to:
\begin{enumerate}
  \item Model physical situations.
  \item Solve the resulting equations.
  \item Interpret the solutions.
\end{enumerate}

\begin{definition}{Ordinary Differential Equation}{ordinary-differential-equation}
  \begin{center}
    \begin{tikzpicture}
      \node (ordinary) at (0,0) {\textbf{Ordinary}};
      \node (differential) at (4,0) {\textbf{Differential}};
      \node (equations) at (8,0) {\textbf{Equations}};
      \node[align=center] (unknown) at (0,-1.2) {One unknown\\function};
      \node[align=center] (derivatives) at (4,-1.2) {$\geq 1$ derivatives};
      \node (equation) at (8,-1.2) {An equation};
      \draw[->] (ordinary) -- (unknown);
      \draw[->] (differential) -- (derivatives);
      \draw[->] (equations) -- (equation);
    \end{tikzpicture}
  \end{center}
  Here, \emph{ordinary} specifies that the unknown function depends on one independent variable.
\end{definition}

\begin{example}[Solving by Integration]
  Solve
  \[
    \frac{dy}{dx}=x^2.
  \]
  Integrating gives
  \[
    y(x)=\frac{x^3}{3}+C, \qquad C\in\R.
  \]
\end{example}

\begin{definition}{Order of an ODE}{ode-order}
  The \textbf{order} of an ODE is the order of the highest derivative appearing in the equation.
\end{definition}

\begin{definition}{Initial Condition}{initial-condition}
  An \textbf{initial condition} specifies a value that a solution (or one of its derivatives) must satisfy at a given point. For example, $y(a)=b$ requires the solution to pass through $(a,b)$.
\end{definition}

\begin{definition}{Initial Value Problem}{initial-value-problem}
  An \textbf{initial value problem (IVP)} consists of an ODE together with initial conditions.
\end{definition}

\begin{example}[Determining the Constant]
  Solve the IVP
  \[
    \begin{cases}
      \dfrac{dy}{dx}=x^2, \\
      y(1)=3.
    \end{cases}
  \]
  The general solution is $y(x)=x^3/3+C$. The initial condition gives
  \[
    3=\frac{1}{3}+C \quad\Longrightarrow\quad C=\frac{8}{3}.
  \]
  Thus
  \[
    y(x)=\frac{x^3}{3}+\frac{8}{3}.
  \]
\end{example}

\subsubsection{Direct Integration}

For an equation of the form
\[
  \frac{dy}{dx}=f(x),
\]
the general solution is an antiderivative of $f$:
\[
  y(x)=F(x)+C, \qquad F'(x)=f(x).
\]
If $f$ is continuous, the IVP
\[
  \begin{cases}
    \dfrac{dy}{dx}=f(x), \\
    y(a)=b
  \end{cases}
\]
has solution
\[
  y(x)=b+\int_a^x f(t)\dd{t}.
\]

\begin{example}[Identifying the Order]
  The equation
  \[
    \frac{d^2y}{dx^2}+3\frac{dy}{dx}+7y=e^x
  \]
  is a second-order ODE, since its highest derivative is $d^2y/dx^2$.
\end{example}

\subsubsection{An Equation Involving the Unknown Function}

\begin{example}[Exponential Growth]
  Solve the IVP
  \[
    \begin{cases}
      \dfrac{dy}{dx}=y, \\
      y(1)=7.
    \end{cases}
  \]
  For $y\ne0$, viewing $x$ as a function of $y$ gives (can also integrate by partial fraction and substitution)
  \[
    \frac{dx}{dy}=\frac{1}{y}.
  \]
  Integrating,
  \[
    x=\ln\abs{y}+C.
  \]
  The initial condition gives $C=1-\ln 7$, so
  \begin{align*}
    x       & = \ln\abs{y}+1-\ln 7,     \\
    \abs{y} & = e^{x+\ln 7-1}=7e^{x-1}.
  \end{align*}
  Since $y(1)=7>0$, the solution is
  \[
    y(x)=7e^{x-1}.
  \]
\end{example}

\subsection{Slope Fields}

For a first-order ODE
\[
  \frac{dy}{dx}=f(x,y),
\]
the value $f(a,b)$ gives the slope of a solution curve passing through $(a,b)$.

\begin{example}[Slope at a Point]
  For $y'=y$, a solution passing through $(1,1)$ has slope $1$ there.
  \begin{center}
    \begin{tikzpicture}[>=latex]
      \draw[->] (-0.6,0) -- (2.3,0) node[right] {$x$};
      \draw[->] (0,-0.5) -- (0,2.2) node[above] {$y$};
      \draw (1,0.06) -- (1,-0.06) node[below] {$1$};
      \draw (0.06,1) -- (-0.06,1) node[left] {$1$};
      \draw[->,thick,blue!70] (0.65,0.65) -- (1.35,1.35);
      \fill (1,1) circle (2pt);
      \node[right] at (1.4,1) {slope $1$};
    \end{tikzpicture}
  \end{center}
\end{example}

\begin{definition}{Slope Field}{slope-field}
  For $y'=f(x,y)$, a \textbf{slope field} assigns the direction
  \[
    \mathbf{F}(x,y)=
    \begin{bmatrix}
      1 \\f(x,y)
    \end{bmatrix}
  \]
  at each point where $f$ is defined. Solution curves are tangent to these directions.
\end{definition}

The plots below use equal-length arrows in the direction $\ve{1,f(x,y)}$ to show slopes clearly.

\subsubsection{Existence and Uniqueness}

\begin{theorem}{Picard's Theorem}{picard}
  Consider the IVP
  \[
    \begin{cases}
      \dfrac{dy}{dx}=f(x,y), \\
      y(a)=b.
    \end{cases}
  \]
  If $f$ and $\pd{f}{y}$ are continuous on a neighborhood of $(a,b)$, then the IVP has a unique solution on some open interval containing $x=a$.
\end{theorem}

These conditions are sufficient, not necessary. The proof is reserved for an upper-division course.

\begin{example}[Applying Picard's Theorem]
  Consider
  \[
    \begin{cases}
      \dfrac{dy}{dx}=2\sin(xy), \\
      y(-1)=1.
    \end{cases}
  \]
  Both
  \[
    f(x,y)=2\sin(xy),\qquad \pd{f}{y}=2x\cos(xy)
  \]
  are continuous everywhere. By Theorem~\ref{thm:picard}, there is a unique local solution passing through $(-1,1)$.
\end{example}

\begin{example}[An Undefined Right-Hand Side]
  Consider
  \[
    \begin{cases}
      \dfrac{dy}{dx}=\dfrac{1}{2y}, \\
      y(0)=0.
    \end{cases}
  \]
  The function $f(x,y)=1/(2y)$ is undefined at $y=0$, so Picard's Theorem (\ref{thm:picard}) does not apply.

  However, the curves $y=\pm\sqrt{x}$ solve the ODE for $x>0$ and approach $(0,0)$.
  \begin{center}

\subsection{Separation of Variables}

\begin{definition}{Separable ODE}{separable-ode}
  An ODE is \textbf{separable} if it can be written in the form
  \[
    \frac{dy}{dx}=g(x)h(y).
  \]
\end{definition}

\subsubsection{Method}

For a separable ODE (Definition~\ref{def:separable-ode}):
\begin{enumerate}
  \item Separate the variables, assuming $h(y)\ne0$:
        \[
          \frac{dy}{h(y)}=g(x)\dd{x}.
        \]
  \item Integrate both sides:
        \[
          \int\frac{1}{h(y)}\dd{y}=\int g(x)\dd{x}.
        \]
  \item Compute the integrals to obtain an implicit solution:
        \[
          H(y)=G(x)+C.
        \]
  \item If possible, solve explicitly for $y$ as a function of $x$.
\end{enumerate}
Check separately for constant solutions at zeros of $h$, since division by $h(y)$ excludes them.

\begin{example}[Justifying Separation by Substitution]
  Consider
  \[
    \frac{dy}{dx}=x^2y^3.
  \]
  The constant function $y(x)=0$ is a solution. For $y\ne0$, divide by $y^3$ and integrate with respect to $x$:
  \[
    y^{-3}\frac{dy}{dx}=x^2
    \quad\Longrightarrow\quad
    \int y^{-3}\frac{dy}{dx}\dd{x}=\int x^2\dd{x}.
  \]
  Substitution on the left gives
  \[
    \int y^{-3}\dd{y}=\int x^2\dd{x}
    \quad\Longrightarrow\quad
    -\frac12 y^{-2}=\frac13 x^3+C.
  \]
\end{example}

\begin{definition}{General and Singular Solutions}{general-singular-solutions}
  A \textbf{general solution} is a family of solutions involving arbitrary constants (one constant for the first-order equations considered here).
  Relative to a given general-solution family, a \textbf{singular solution} is a solution that cannot be obtained by choosing values of those constants.
\end{definition}

In the preceding example, $-\tfrac12 y^{-2}=\tfrac13 x^3+C$ is the general solution, and $y(x)=0$ is a singular solution in the sense of Definition~\ref{def:general-singular-solutions}.

\begin{example}[Absorbing a Constant Solution into the Family]
  Solve
  \[
    \frac{dy}{dx}=3x^2y.
  \]
  Either $y(x)=0$, or separation gives
  \begin{align*}
    \int y^{-1}\dd{y} & = \int 3x^2\dd{x},                \\
    \ln\abs{y}        & = x^3+C,                          \\
    \abs{y}           & = e^{x^3+C}=Ae^{x^3}, \qquad A>0.
  \end{align*}
  Allowing either sign gives $y=Ae^{x^3}$ for $A\ne0$. Including $A=0$ incorporates the constant solution, so
  \[
    y(x)=Ae^{x^3},\qquad A\in\R.
  \]
\end{example}

\begin{example}[Spotting a Solution to an IVP]
  Find a particular solution of
  \[
    \begin{cases}
      \dfrac{dy}{dx}=y^3\tan(x^2), \\
      y(1)=0.
    \end{cases}
  \]
  By inspection, $y(x)=0$ is a solution on an interval about $x=1$ where $\tan(x^2)$ is defined.
  Both $f(x,y)=y^3\tan(x^2)$ and $\pd{f}{y}=3y^2\tan(x^2)$ are continuous near $(1,0)$, so Picard's Theorem (\ref{thm:picard}) guarantees local uniqueness.
\end{example}

\begin{example}[Partial Fractions and the Choice of Parameter]
  Solve
  \[
    \frac{dy}{dx}=2xy(y-1).
  \]
  First note the constant solutions $y=0$ and $y=1$. For $y\ne0,1$, separate variables and use partial fractions:
  \begin{align*}
    \int\frac{1}{y(y-1)}\dd{y}                       & = \int 2x\dd{x}, \\
    \int\left(\frac{1}{y-1}-\frac{1}{y}\right)\dd{y} & = \int 2x\dd{x}, \\
    \ln\abs{y-1}-\ln\abs{y}                          & = x^2+C,         \\
    \ln\abs{\frac{y-1}{y}}                           & = x^2+C,         \\
    \frac{y-1}{y}                                    & = Ae^{x^2}.
  \end{align*}
  Hence
  \[
    y(x)=\frac{1}{1-Ae^{x^2}}.
  \]
  Extending the parameter to $A=0$ includes $y=1$, whereas $y=0$ is singular relative to this family.

  Alternatively, for $A\ne0$, let $B=A^{-1}$. Then
  \[
    y(x)=\frac{1}{1-B^{-1}e^{x^2}}
    =\frac{B}{B-e^{x^2}}.
  \]
  Extending the latter family to $B=0$ includes $y=0$, whereas $y=1$ is singular relative to this family. In each form, solutions are taken on intervals where the denominator is nonzero.

  Thus the term \emph{singular solution}, as used here, is meaningful only relative to a specified general-solution family.
\end{example}

\subsection{Linear First-Order Equations}

\begin{definition}{Linear First-Order ODE}{linear-first-order-ode}
  A \textbf{linear first-order ODE} has the form
  \[
    \frac{dy}{dx}+p(x)y=f(x),
  \]
  where $p$ and $f$ are given functions. The unknown function $y$ and its derivative $y'$ appear linearly.
\end{definition}

\begin{example}[Recognizing Linearity]
  The equation $y'+17x^2y=e^x$ is linear and first-order.
  The equation $y'+y^2=3$ is first-order but is not linear.
\end{example}

\subsubsection{The Integrating Factor Method}

\begin{example}[Turning the Left-Hand Side into a Product Derivative]
  Solve $y'+7y=x$. Multiply both sides by $e^{7x}$:
  \[
    e^{7x}y'+7e^{7x}y=xe^{7x}.
  \]
  By the product rule, the left-hand side is $\dfrac{d}{dx}(e^{7x}y)$, so
  \[
    \frac{d}{dx}(e^{7x}y)=xe^{7x}.
  \]
  Integrating by parts gives
  \begin{align*}
    e^{7x}y & = \int xe^{7x}\dd{x}                       \\
            & = \frac17 xe^{7x}-\int\frac17 e^{7x}\dd{x} \\
            & = \frac17 xe^{7x}-\frac1{49}e^{7x}+C.
  \end{align*}
  Thus
  \[
    y(x)=\frac{x}{7}-\frac1{49}+Ce^{-7x}.
  \]
\end{example}

\begin{definition}{Integrating Factor}{integrating-factor}
  For $y'+p(x)y=f(x)$, choose an \textbf{integrating factor}
  \[
    r(x)=e^{\int p(x)\dd{x}},
  \]
  using any one antiderivative of $p$ in the exponent.
\end{definition}

To justify the method, differentiate the integrating factor from Definition~\ref{def:integrating-factor}:
\[
  r'(x)=\frac{d}{dx}e^{\int p(x)\dd{x}}
  =p(x)e^{\int p(x)\dd{x}}=p(x)r(x).
\]
Multiplying the ODE by $r(x)$ therefore yields
\[
  ry'+rpy=ry'+r'y=\frac{d}{dx}(ry)=r(x)f(x).
\]
Integrate and divide by $r(x)$ to obtain
\[
  r(x)y=\int r(x)f(x)\dd{x}+C,
  \qquad
  y(x)=\frac{1}{r(x)}\left(\int r(x)f(x)\dd{x}+C\right).
\]

\begin{example}[A Variable Coefficient]
  Solve
  \[
    \frac{dy}{dx}-3x^2y=x^2.
  \]
  The integrating factor is
  \[
    r(x)=e^{\int -3x^2\dd{x}}=e^{-x^3}.
  \]
  Thus
  \begin{align*}
    e^{-x^3}y & = \int e^{-x^3}x^2\dd{x}+C
    =-\frac13 e^{-x^3}+C,                                 \\
    y(x)      & = e^{x^3}\left(-\frac13 e^{-x^3}+C\right)
    =-\frac13+Ce^{x^3}.
  \end{align*}
\end{example}

\begin{example}[An Initial Value Problem]
  Solve
  \[
    \begin{cases}
      xy'+7y=x^2, \\
      y(1)=2.
    \end{cases}
  \]
  Divide by $x$ to put the equation in standard form:
  \[
    y'+\frac7x y=x.
  \]
  An integrating factor is
  \[
    r(x)=e^{\int (7/x)\dd{x}}=e^{7\ln\abs{x}}=\abs{x}^7.
  \]
  We work on $x>0$, the interval containing the initial point $x=1$ on which the coefficients are continuous. Hence $r(x)=x^7$, and
  \[
    y(x)=\frac{1}{x^7}\left(\int x^8\dd{x}+C\right)
    =\frac{x^2}{9}+Cx^{-7}.
  \]
  The initial condition gives $2=\tfrac19+C$, so $C=\tfrac{17}{9}$ and
  \[
    y(x)=\frac19\left(x^2+17x^{-7}\right),\qquad x>0.
  \]
  Alternatively, incorporate the initial condition directly using a definite integral:
  \[
    y(x)=\frac{1}{x^7}\left(\int_1^x t^8\dd{t}+2\right).
  \]
\end{example}

\begin{example}[The Reservoir Problem]
  A reservoir has volume $W_0=160{,}000$ and initial pollution concentration
  $P_0=0.25\%$. Every day, $W_1=10{,}000$ of water with concentration
  $P_1=0.05\%$ flows in, is well mixed, and $10{,}000$ flows out.
  When is the concentration $P=0.1\%$?

  \begin{center}
    \begin{tikzpicture}[>=latex]
      \fill[blue!8] (0,0) rectangle (4,1.5);
      \draw[thick] (0,2) -- (0,0) -- (4,0) -- (4,2);
      \draw[blue!60] (0,1.5) -- (4,1.5);
      \node[align=center] at (2,0.75) {$W_0=160{,}000$\\well mixed};
      \draw[->,thick] (-1.7,2.1) -- (0.6,2.1) -- (0.6,1.6);
      \node[align=center,above] at (-0.4,2.1) {$10{,}000$/day\\$P_1=0.05\%$};
      \draw[->,thick] (4,0.5) -- (7.3,0.5);
      \node[align=left,anchor=west] at (4.3,1) {$10{,}000$/day\\$P(t)=V(t)/W_0$};
    \end{tikzpicture}
  \end{center}

  Let $V(t)$ be the volume of pollutant in the reservoir at $t$ days. Then
  \[
    V=PW_0,\qquad V(0)=P_0W_0=400.
  \]
  Over a short time $dt$, treat $V$ as constant. The rate in minus the rate out gives
  \[
    \frac{dV}{dt}=P_1W_1-\frac{V}{W_0}W_1
    =0.05\%\times10{,}000-\frac{V}{160{,}000}\,10{,}000
    =5-\frac{V}{16}.
  \]
  Thus the IVP is
  \[
    \begin{cases}
      V'+\dfrac1{16}V=5, \\
      V(0)=400.
    \end{cases}
  \]
  With integrating factor $r(t)=e^{t/16}$,
  \[
    V(t)=e^{-t/16}\left(\int 5e^{t/16}\dd{t}+C\right)
    =80+Ce^{-t/16}=80+320e^{-t/16}.
  \]
\end{example}

\subsection{Autonomous Equations}

\begin{definition}{Autonomous First-Order ODE}{autonomous-first-order-ode}
  An equation of the form
  \[
    \frac{dx}{dt}=f(x),
  \]
  with right-hand side independent of $t$, is an \textbf{autonomous first-order ODE}.
\end{definition}

Always separable: where $f(x)\ne0$,
\[
  \int\frac{1}{f(x)}\dd{x}=t+C.
\]

\begin{definition}{Equilibrium Solution}{equilibrium-solution}
  If $f(c)=0$, then $x(t)\equiv c$ is a constant or \textbf{equilibrium solution}.
  The value $c$ is a \textbf{critical point}.
\end{definition}

\begin{definition}{Stable and Unstable Equilibria}{equilibrium-stability}
  An equilibrium $x(t)\equiv c$ is \textbf{stable} if every solution starting
  sufficiently near $c$ stays near $c$ as $t$ increases. Otherwise, it is \textbf{unstable}.
\end{definition}

\begin{example}[A Cubic Autonomous Equation]
  Consider
  \[
    \frac{dx}{dt}=x(x-1)(x-3).
  \]
  The equilibrium solutions are $x(t)\equiv0,1,3$.
  The equilibrium $1$ is stable; $0$ and $3$ are unstable.

      \begin{axis}[

\begin{example}[A Sine Autonomous Equation]
  Consider $dx/dt=\sin x$. Critical points and stability:
  \[
    \sin x=0\quad\Longrightarrow\quad x=n\pi,\qquad n\in\Z.
  \]
  The equilibrium $x(t)\equiv n\pi$ is stable for odd $n$ and unstable for even $n$.
  \begin{center}
    \begin{tikzpicture}
      \begin{axis}[
        width=11cm,height=9cm,
        axis lines=left,xlabel={$t$},ylabel={$x$},
        xmin=0,xmax=6,ymin=-2.2,ymax=3.95,
        xtick={0,2,4,6},ytick={-2,-1,0,1,2,3},
        yticklabels={$-2\pi$,$-\pi$,$0$,$\pi$,$2\pi$,$3\pi$},
        clip=false,
      ]
        \pgfplotsextra{
          \foreach \t in {0.25,0.75,...,5.75} {
            \foreach \z in {-2,-1.75,...,3.75} {
              % Since z=x/pi, dz/dt=sin(pi*z)/pi.
              \pgfmathsetmacro{\slope}{sin(deg(pi*\z))/pi}
              \pgfmathsetmacro{\dt}{0.12/sqrt(1+\slope^2)}
              \pgfmathsetmacro{\dz}{\slope*\dt}
              \draw[black!20,thin]
                (axis cs:{\t-\dt},{\z-\dz}) --
                (axis cs:{\t+\dt},{\z+\dz});
            }
          }
        }
        % Vertical coordinates are x/pi; these are exact solution curves.
        \foreach \c in {-1,1,3} {
          \foreach \d in {-0.9,0.9} {
            \addplot[blue!75!black,thick,domain=0:6,samples=120]
              {\c+2/pi*rad(atan(tan(deg(\d*pi/2))*exp(-x)))};
          }
          \addplot[green!40!black,thick,domain=0:6] {\c};
        }
        \foreach \c in {-2,0,2} {
          \addplot[red!75!black,thick,domain=0:6] {\c};
        }
        \node[right,green!40!black] at (axis cs:6,-1) {stable};
        \node[right,green!40!black] at (axis cs:6,1) {stable};
        \node[right,green!40!black] at (axis cs:6,3) {stable};
        \node[right,red!75!black] at (axis cs:6,-2) {unstable};
        \node[right,red!75!black] at (axis cs:6,0) {unstable};
        \node[right,red!75!black] at (axis cs:6,2) {unstable};
      \end{axis}
    \end{tikzpicture}
  \end{center}
\end{example}

\subsubsection{Population Models}

\begin{example}[Natural Growth and Decay]
  Suppose the population is $P(t)$, with birth rate $\beta$ and death rate $\delta$
  (per member per unit time). Then
  \[
    \frac{dP}{dt}=(\beta-\delta)P.
  \]
  If $\beta$ and $\delta$ are constant, let $k=\beta-\delta$. Separation gives
  \[
    \frac{dP}{dt}=kP,\qquad \ln\abs{P}=kt+C,\qquad P(t)=P_0e^{kt}.
  \]
\end{example}

\begin{example}[The Logistic Model]
  Suppose $\beta-\delta=k(M-P(t))$, a decreasing linear function of $P(t)$.
  Then
  \[
    \frac{dP}{dt}=kP(M-P).
  \]
\end{example}
