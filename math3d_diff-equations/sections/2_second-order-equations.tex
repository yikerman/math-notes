\subsection{Linear Second-Order Equations}

\begin{definition}{Second-Order ODE}{second-order-ode}
  A \textbf{second-order ODE} has highest derivative $d^2y/dx^2$.
  A \textbf{linear second-order ODE} has the form
  \[
    y''+p(x)y'+q(x)y=f(x).
  \]
  It is \textbf{homogeneous} if $f(x)=0$.
\end{definition}

\begin{example}[Classifying Second-Order Equations]
  The equation
  \[
    \frac{d^2y}{dx^2}+(\sin x)\frac{dy}{dx}+e^{3x^2}y=4\tan x
  \]
  is a nonhomogeneous linear second-order differential equation.
  The equation $y''+yy'=0$ is nonlinear and second-order.
\end{example}

\begin{theorem}{Existence and Uniqueness for Linear Second-Order Equations}{linear-second-order-existence-uniqueness}
  Consider the IVP
  \[
    \begin{cases}
      y''+p(x)y'+q(x)y=f(x), \\
      y(a)=b_0,\qquad y'(a)=b_1.
    \end{cases}
  \]
  If $p$, $q$, and $f$ are continuous on an interval $I$ containing $a$,
  then the IVP has a unique solution on $I$.
\end{theorem}

\begin{example}[Solutions by Inspection]
  For $y''-4y=0$, the functions $e^{2x}$ and $e^{-2x}$ are solutions.
  Every solution has the form
  \[
    y(x)=C_1e^{2x}+C_2e^{-2x}.
  \]
\end{example}

\begin{definition}{Independent Solutions}{independent-solutions}
  Two nonzero solutions $y_1$ and $y_2$ are \textbf{independent} if neither
  is a constant multiple of the other.
\end{definition}

\begin{theorem}{Superposition Theorem}{superposition}
  Suppose $p$ and $q$ are continuous on an interval $I$, and $y_1$, $y_2$
  are independent solutions (Definition~\ref{def:independent-solutions}) of
  \[
    y''+p(x)y'+q(x)y=0.
  \]
  Then all solutions on $I$ have the form
  \[
    y(x)=C_1y_1(x)+C_2y_2(x),\qquad C_1,C_2\in\R.
  \]
\end{theorem}

\begin{example}[A Power Solution]
  Consider $x^2y''+2xy'-6y=0$.

  \begin{thinking}
    Guess $y=x^r$, $r$ is constant.\\
    Since $y'$ reduces the power,\\
    the coefficient multiplies $x$ back.\\
    Make the LHS the same order of $x$.
  \end{thinking}

  \begin{align*}
    x^2(x^r)''+2x(x^r)'-6x^r         & =0,      \\
    x^2r(r-1)x^{r-2}+2xrx^{r-1}-6x^r & \equiv0, \\
    r(r-1)+2r-6                      & =0,      \\
    (r-2)(r+3)                       & =0.
  \end{align*}
  Since the expression is $0$ regardless of $x$, the coefficient has to be $0$.
  \[
    r_1=2,\qquad r_2=-3.
  \]
  General solution on an interval not containing $0$:
  \[
    y(x)=C_1x^2+C_2x^{-3},\qquad \forall C_1,C_2\in\R.
  \]
\end{example}

\subsection{Constant-Coefficient Second-Order Linear ODEs}

Given constants $a,b,c\in\R$, $a\ne0$,
\[
  ay''+by'+cy=0.
\]
\begin{thinking}
  Guess $y=e^{\lambda x}$. (Hint: eigenvalue! More later.)
  \[
    (a\lambda^2+b\lambda+c)e^{\lambda x}\equiv0.
  \]
  Since $\equiv$ means equality as functions,
  \[
    a\lambda^2+b\lambda+c=0.
  \]
\end{thinking}

\begin{definition}{Characteristic Equation}{characteristic-equation}
  The \textbf{characteristic equation} of $ay''+by'+cy=0$ is
  \[
    a\lambda^2+b\lambda+c=0.
  \]
\end{definition}

$y=e^{\lambda x}$ is a solution to the ODE $\Longleftrightarrow$
$\lambda$ is a root of the characteristic equation
(Definition~\ref{def:characteristic-equation}).

\subsubsection{Case 1: Distinct Real Roots}

$\lambda_1\ne\lambda_2$ are real numbers.

\begin{example}[Distinct Roots and an IVP]
  \[
    y''-2y'-3y=0.
  \]
  Characteristic equation:
  \begin{align*}
    \lambda^2-2\lambda-3   & =0, \\
    (\lambda-3)(\lambda+1) & =0.
  \end{align*}
  General solution:
  \[
    y(x)=C_1e^{3x}+C_2e^{-x}.
  \]
  Suppose $y(0)=7$, $y'(0)=1$ (IVP). Need $C_1,C_2$ by
  \[
    \begin{cases}
      C_1+C_2=7  & \text{evaluate }y(0),  \\
      3C_1-C_2=1 & \text{evaluate }y'(0).
    \end{cases}
  \]
  \[
    4C_1=8\implies C_1=2,\ C_2=5
    \implies y(x)=2e^{3x}+5e^{-x}.
  \]
\end{example}

\subsubsection{Case 2: Repeated Root}

$\lambda_1=\lambda_2$.

\begin{example}[A Repeated Root]
  \[
    y''-6y'+9y=0.
  \]
  Characteristic equation:
  \begin{align*}
    \lambda^2-6\lambda+9 & =0, \\
    (\lambda-3)^2        & =0, \\
    \lambda_1=\lambda_2  & =3.
  \end{align*}
  Theory says $y_1=e^{3x}$.

  \begin{thinking}
    What if we let $y_2=xy_1=xe^{3x}$?
    \begin{align*}
      y_2'  & =e^{3x}+3xe^{3x}=(1+3x)e^{3x}, \\
      y_2'' & =(6+9x)e^{3x}.
    \end{align*}
    Then
    \[
      y_2''-6y_2'+9y_2
      =(6+9x-6(1+3x)+9x)e^{3x}\equiv0.
    \]
  \end{thinking}

  General solution:
  \[
    y(x)=C_1e^{3x}+C_2xe^{3x}.
  \]
\end{example}

Generally, if the characteristic equation is $(\lambda-t)^2=0$
$\Longleftrightarrow\lambda_1=\lambda_2=t$, then
\[
  y=(C_1+C_2x)e^{tx}.
\]

\begin{thinking}[Proof Sketch]
  \[
    ay''+by'+cy=0.
  \]
  Characteristic equation:
  \[
    a\lambda^2+b\lambda+c=0.
  \]
  \[
    \Delta=0\Longleftrightarrow b^2=4ac,
    \qquad \lambda=-\frac{b}{2a}.
  \]
\end{thinking}

\subsubsection{Case 3: Complex Conjugate Roots}

The characteristic equation has solutions
\[
  \lambda_1=p+iq,\qquad \lambda_2=p-iq=\overline{\lambda_1},\qquad q\ne0.
\]
Conclusion:
\[
  y_1=e^{px}\cos qx\quad\text{and}\quad y_2=e^{px}\sin qx
\]
are independent solutions.

\begin{thinking}
  Or:
  \begin{align*}
    e^{(p+iq)x} & =e^{px}\cos qx+ie^{px}\sin qx, \\
    e^{(p-iq)x} & =e^{px}\cos qx-ie^{px}\sin qx.
  \end{align*}
  Different bases for the same complex vector subspace.
\end{thinking}

\begin{example}[Complex Roots]
  \[
    y''-3y'+10y=0.
  \]
  \[
    \lambda^2-3\lambda+10=0,
    \qquad \lambda=\frac{3\pm\sqrt{31}\,i}{2}
    =\frac32\pm\frac{\sqrt{31}}2i.
  \]
  General solution:
  \begin{align*}
    y & =C_1\left(e^{3x/2}\cos\frac{\sqrt{31}}2x\right)
    +C_2\left(e^{3x/2}\sin\frac{\sqrt{31}}2x\right)                                \\
      & =\left(C_1\cos\frac{\sqrt{31}}2x+C_2\sin\frac{\sqrt{31}}2x\right)e^{3x/2}.
  \end{align*}
  \begin{center}

    \end{tikzpicture}
  \end{center}
\end{example}

\subsection{Higher-Order Linear ODEs}

\begin{example}[Three Distinct Real Roots]
  \[
    y'''-2y''-5y'+6y=0.
  \]
  Characteristic equation ($y^{(k)}\mapsto\lambda^k$):
  \[
    \lambda^3-2\lambda^2-5\lambda+6=0.
  \]
  Guess one root:
  \begin{align*}
    (\lambda-1)(\lambda^2-\lambda-6)  & =0, \\
    (\lambda-1)(\lambda-3)(\lambda+2) & =0.
  \end{align*}
  General solution:
  \[
    y=C_1e^x+C_2e^{3x}+C_3e^{-2x}.
  \]
\end{example}

For $\sum_{k=0}^n a_k y^{(k)}=0$, with real constant coefficients and $a_n\ne0$,
\[
  y^{(k)}\mapsto\lambda^k,\qquad\text{solve the characteristic equation.}
\]
For a real root $\lambda$, get an independent solution $y=e^{\lambda x}$.

For complex roots $\lambda=p\pm iq$,
\[
  y=e^{px}\cos qx\quad\text{and}\quad y=e^{px}\sin qx.
\]
For a repeated real root $\lambda_1=\lambda_2=\lambda$,
\[
  y=e^{\lambda x}\quad\text{and}\quad y=xe^{\lambda x}.
\]

\begin{example}[A Double Root]
  \[
    y'''-3y'+2y=0.
  \]
  Characteristic equation:
  \begin{align*}
    \lambda^3-3\lambda+2              & =0, \\
    (\lambda-1)(\lambda^2+\lambda-2)  & =0, \\
    (\lambda-1)(\lambda-1)(\lambda+2) & =0.
  \end{align*}
  \[
    \lambda_1=\lambda_2=1,\qquad\lambda_3=-2.
  \]
  General solution:
  \[
    y=C_1e^x+C_2xe^x+C_3e^{-2x}.
  \]
\end{example}

\begin{example}[A Triple Root]
  \[
    y'''-3y''+3y'-y=0.
  \]
  Characteristic equation: $(\lambda-1)^3=0$.

  General solution:
  \[
    y=(C_1+C_2x+C_3x^2)e^x.
  \]
\end{example}

\begin{example}[Two Pairs of Complex Roots]
  \[
    y^{(4)}+5y''+4y=0.
  \]
  Characteristic equation:
  \begin{align*}
    \lambda^4+5\lambda^2+4     & =0, \\
    (\lambda^2+1)(\lambda^2+4) & =0.
  \end{align*}
  \[
    \lambda_1=i,\quad\lambda_2=-i,\quad\lambda_3=2i,\quad\lambda_4=-2i.
  \]
  General solution:
  \[
    y(x)=C_1\cos x+C_2\sin x+C_3\cos2x+C_4\sin2x.
  \]
\end{example}

\begin{example}[Repeated Complex Roots]
  \[
    y^{(4)}+2y''+y=0.
  \]
  Characteristic equation:
  \begin{align*}
    \lambda^4+2\lambda^2+1 & =0, \\
    (\lambda^2+1)^2        & =0.
  \end{align*}
  $\lambda_1=\lambda_2=i$, $\lambda_3=\lambda_4=-i$: repeated complex roots.

  General solution:
  \[
    y=C_1\cos x+C_2\sin x+C_3x\cos x+C_4x\sin x.
  \]
\end{example}

\begin{example}[Find the ODE]
  Suppose
  \begin{align*}
    y={} & (C_1+C_2x+C_3x^2)e^{-2x}\cos3x   \\
         & +(C_4+C_5x+C_6x^2)e^{-2x}\sin3x.
  \end{align*}
  Find the ODE.

  Roots: $\lambda=-2\pm3i$, multiplicity $3$ each.

  Characteristic equation:
  \[
    (\lambda+2-3i)^3(\lambda+2+3i)^3
    =(\lambda^2+4\lambda+13)^3=0.
  \]
  \[
    \lambda^6+\cdots=0
    \quad\longrightarrow\quad y^{(6)}+\cdots=0.
  \]
\end{example}

\subsubsection{Theorems}

\begin{theorem}{Picard's Theorem: Specialized Case}{higher-order-existence-uniqueness}
  A linear $n$th-order ODE with initial conditions
  \[
    \begin{cases}
      \displaystyle\sum_{i=0}^n P_i(x)y^{(i)}=f(x), \\
      y(a)=b_0,\quad y'(a)=b_1,\quad\ldots,\quad y^{(n-1)}(a)=b_{n-1}
    \end{cases}
  \]
  has a unique solution on an interval $I$ containing $a$, if $P_0,\ldots,P_n,f$
  are continuous on $I$ and $P_n(x)\ne0$ on $I$.
\end{theorem}

\begin{theorem}{Superposition Theorem: Specialized Case}{higher-order-superposition}
  If $y_1,y_2$ are solutions to the homogeneous $n$th-order linear ODE
  \[
    \sum_{i=0}^n P_i(x)y^{(i)}=0,
  \]
  then $C_1y_1+C_2y_2$ is also a solution.
\end{theorem}

\begin{definition}{Linearly Independent Solutions}{higher-order-independent-solutions}
  Recall: solutions $y_1,\ldots,y_n$ are \textbf{independent} if
  \[
    C_1y_1(x)+\cdots+C_ny_n(x)\equiv0
    \implies C_1=\cdots=C_n=0.
  \]
\end{definition}

Under the coefficient assumptions of Theorem~\ref{thm:higher-order-existence-uniqueness},
if $y_1,\ldots,y_n$ are independent solutions of the homogeneous equation, the general solution is
\[
  y=C_1y_1+\cdots+C_ny_n.
\]

\subsection{Mechanical Vibrations}

Model a spring attached to a fixed point and mass $m$.
\begin{center}
  \begin{tikzpicture}[>=latex]
    \draw[thick] (0,-0.7) -- (0,0.7);
    \draw[thick] (0,0) -- (0.4,0);
    \draw[thick,domain=0:6*pi,samples=180,variable=\t]
      plot ({0.4+0.15*\t-0.1*sin(\t r)},{0.18*sin(\t r)});
    \draw[thick] ({0.4+0.9*pi},0) -- (3.6,0);
    \draw[thick] (3.6,-0.25) rectangle (4.1,0.25);
    \fill (3.85,0) circle (1.5pt);
    \node[below] at (3.85,-0.25) {$m$};
    \draw[<->] (3.4,-1) -- (4.3,-1) node[right] {$x(t)$};
  \end{tikzpicture}
\end{center}
$x(t)$: extension of spring from equilibrium.

Forces on mass:
\begin{itemize}
  \item Spring force: $F_S=-kx$ ($k>0$ constant, ``stiffness'').
  \item Friction: $F_R=-cx'$ ($c\ge0$; magnitude proportional to speed).
        (Actually, a bad model.)
  \item External: $F_E=$ anything else\ldots?
\end{itemize}
Apply $F=ma$:
\begin{align*}
  mx''        & =-kx-cx'+F_E, \\
  mx''+cx'+kx & =F_E.
\end{align*}

\begin{definition}{Free Motion}{free-motion}
  For $F_E=0$,
  \[
    mx''+cx'+kx=0
  \]
  is homogeneous: called \textbf{free} or \textbf{unforced motion}.
\end{definition}
Easy to solve!

\subsubsection{Simple Harmonic Motion}

Easiest: $c=0$.
\[
  m\lambda^2+k=0\implies\lambda=\pm i\sqrt{\frac{k}{m}}.
\]
Let $\omega_0=\sqrt{k/m}$ be the natural frequency of the spring.

General solution:
\[
  x=A\cos\omega_0t+B\sin\omega_0t
  \qquad\text{(simple harmonic motion).}
\]
Period: $T=2\pi/\omega_0$, $f=1/T$.

\subsubsection{Free Damped Motion}

\[
  mx''+cx'+kx=0,\qquad c>0.
\]
Characteristic equation:
\[
  m\lambda^2+c\lambda+k=0,
  \qquad\lambda=\frac{-c\pm\sqrt{c^2-4mk}}{2m}.
\]
\begin{enumerate}
  \item \textbf{Overdamping}: $c$ is big, $c^2>4km$.

        Two real negative roots:
        \[
          0<c^2-4km<c^2\implies-c+\sqrt{c^2-4km}<0.
        \]
        Let $\lambda=-r_1,-r_2$ ($r_1,r_2>0$).
        \[
          x(t)=C_1e^{-r_1t}+C_2e^{-r_2t}.
        \]
  \item \textbf{Critical damping}: $c^2=4km$.
        \[
          \lambda=-\frac{c}{2m}\qquad\text{(two equal roots).}
        \]
        Let $p=c/(2m)$, so $\lambda=-p$.
        \[
          x(t)=(a+bt)e^{-pt}.
        \]
  \item \textbf{Underdamping}: $c^2<4km$.
        \begin{align*}
          \lambda & =-\frac{c}{2m}\pm i\frac{\sqrt{4km-c^2}}{2m} \\
                  & =-p\pm i\sqrt{\frac{k}{m}-\frac{c^2}{4m^2}}.
        \end{align*}
        Let
        \[
          \omega_1=\sqrt{\frac{k}{m}-\frac{c^2}{4m^2}}
          =\sqrt{\omega_0^2-\frac{c^2}{4m^2}}<\omega_0.
        \]
        \[
          x(t)=Ae^{-pt}\cos\omega_1t+Be^{-pt}\sin\omega_1t,
        \]
        or
        \[
          x(t)=Re^{-pt}\cos(\omega_1t-\gamma),
          \qquad\gamma:\text{ phase angle}.
        \]
\end{enumerate}
\begin{center}
  \begin{tikzpicture}[>=latex]
    \draw[->] (-0.4,0) -- (7,0) node[right] {$t$};
    \draw[->] (0,-1.6) -- (0,1.8) node[above] {$x$};
    \draw[dashed,gray,domain=0:6.6,samples=80]
      plot (\x,{1.4*exp(-0.4*\x)});
    \draw[dashed,gray,domain=0:6.6,samples=80]
      plot (\x,{-1.4*exp(-0.4*\x)});

  \end{tikzpicture}
\end{center}
\begin{enumerate}
  \item Sine wave asymptotically decays to $0$.
  \item Frequency $\omega_1$ less than $\omega_0$.
\end{enumerate}

\subsection{Nonhomogeneous Equations}

\begin{definition}{Linear Nonhomogeneous Equation}{linear-nonhomogeneous-equation}
  A linear nonhomogeneous equation has the form
  \[
    y^{(n)}+p_{n-1}(x)y^{(n-1)}+\cdots+p_1(x)y'+p_0(x)y=f(x),
  \]
  where $f\not\equiv0$.
\end{definition}

Suppose $y_1$ and $y_2$ are both solutions to the equation above.
Let $z(x)=y_1(x)-y_2(x)$.

Since differentiation is linear:
\[
  \forall k\le n,\qquad z^{(k)}(x)=y_1^{(k)}(x)-y_2^{(k)}(x).
\]
\begin{align*}
  z^{(n)}+p_{n-1}z^{(n-1)}+\cdots+p_0z
   & =y_1^{(n)}+p_{n-1}y_1^{(n-1)}+\cdots+p_0y_1                  \\
   & \quad-\bigl(y_2^{(n)}+p_{n-1}y_2^{(n-1)}+\cdots+p_0y_2\bigr) \\
   & =f(x)-f(x)=0.
\end{align*}

\begin{theorem}{General Solution of a Nonhomogeneous Equation}{nonhomogeneous-general-solution}
  If $y_c$ is the general solution to the corresponding linear homogeneous
  equation and $y_p$ is any particular solution to the nonhomogeneous equation,
  then the nonhomogeneous equation has general solution
  \[
    y(x)=\underbrace{y_c(x)}_{\text{complementary}}
    +\underbrace{y_p(x)}_{\text{particular}}.
  \]
\end{theorem}

\begin{example}[Constant Right-Hand Side]
  Solve $y''+4y=1$.
  \begin{enumerate}
    \item $\lambda^2+4=0$, $\lambda=\pm2i$.
          \[
            y_c(x)=C_1\cos2x+C_2\sin2x.
          \]
    \item Guess $y_p(x)=\frac14$.
    \item General solution:
          \[
            y(x)=y_c+y_p=C_1\cos2x+C_2\sin2x+\frac14.
          \]
  \end{enumerate}
\end{example}

\begin{example}[Exponential Right-Hand Side]
  Find the general solution to $y''+3y'+2y=e^x$.
  \begin{enumerate}
    \item For the homogeneous equation:
          \[
            \lambda^2+3\lambda+2=0,\qquad\lambda_1=-1,\quad\lambda_2=-2.
          \]
          \[
            y_c(x)=C_1e^{-x}+C_2e^{-2x}.
          \]
    \item Trick: guess something like the right side and its derivatives.

          Guess $y_p=ae^x$.
          \[
            y_p''+3y_p'+2y_p=e^x.
          \]
          Solve: $y_p=\frac16e^x$.
    \item General solution:
          \[
            y(x)=C_1e^{-x}+C_2e^{-2x}+\frac16e^x.
          \]
  \end{enumerate}
\end{example}

\begin{example}[A Guess Already in the Complementary Solution]
  \[
    y''+3y'+2y=e^{-x}.
  \]
  \begin{enumerate}
    \item $y_c=C_1e^{-x}+C_2e^{-2x}$.
    \item Guess $y_p=ae^{-x}$: same basis as $y_c$.

          $\implies$ Guess $y_p=axe^{-x}$.
          \begin{align*}
            y_p'  & =ae^{-x}(1-x),                 \\
            y_p'' & =ae^{-x}(-1+x-1)=ae^{-x}(x-2).
          \end{align*}
          \begin{align*}
            y_p''+3y_p'+2y_p                 & =e^{-x},         \\
            ae^{-x}\bigl(x-2+3(1-x)+2x\bigr) & =ae^{-x}=e^{-x}, \\
            a                                & =1.
          \end{align*}
          $y_p=xe^{-x}$.
    \item $y(x)=C_1e^{-x}+C_2e^{-2x}+xe^{-x}$.
  \end{enumerate}
\end{example}

\begin{example}[Choosing Particular-Solution Guesses]
  \[
    y'''-3y''+3y'-y=f(x).
  \]
  \begin{enumerate}
    \item $y_c=(C_1+C_2x+C_3x^2)e^x$.
    \item For $f(x)=e^x$:
          \[
            ae^x\;\text{\xmark},\qquad axe^x\;\text{\xmark},\qquad
            ax^2e^x\;\text{\xmark}.
          \]
          Guess $y_p=ax^3e^x$.
          \[
            \begin{array}{ll}
              f(x)=3x^2+7  & : y_p=ax^2+bx+c,          \\[4pt]
              f(x)=2\cos3x & : y_p(x)=a\cos3x+b\sin3x.
            \end{array}
          \]
  \end{enumerate}
\end{example}

\begin{example}[Nonhomogeneous IVP]
  \[
    \begin{cases}
      y''+y=e^x, \\
      y(0)=1,    \\
      y'(0)=-2.
    \end{cases}
  \]
  \begin{enumerate}
    \item $y_c=C_1\cos x+C_2\sin x$.
    \item Guess $y_p=ae^x$.

          Sub:
          \[
            (ae^x)''+ae^x=e^x,\qquad a=\frac12.
          \]
          \[
            y_p=\frac12e^x.
          \]
    \item $y(x)=y_c+y_p=C_1\cos x+C_2\sin x+\frac12e^x$.
    \item IC:
          \[
            y(0)=C_1+\frac12=1,\qquad C_1=\frac12.
          \]
          \begin{align*}
            y'(x) & =-C_1\sin x+C_2\cos x+\frac12e^x,    \\
            y'(0) & =C_2+\frac12=-2,\qquad C_2=-\frac52.
          \end{align*}
          \[
            y(x)=\frac12\bigl(\cos x-5\sin x+e^x\bigr).
          \]
  \end{enumerate}
\end{example}

\subsection{Forced Oscillations}

\[
  mx''+cx'+kx=F_E.
\]

\begin{example}[Damped Driven Oscillation]
  \[
    x''+3x'+2x=F_0\cos\omega t.
  \]
  \begin{enumerate}
    \item Characteristic equation:
          \[
            \lambda^2+3\lambda+2=0
            \qquad\text{(overdamped: $c^2>4km$).}
          \]
          \[
            \lambda_1=-1,\qquad\lambda_2=-2.
          \]
          \[
            x_c(t)=C_1e^{-t}+C_2e^{-2t}.
          \]
          ($x_c$ is transient: $x_c\to0$.)
    \item Guess $x_p=a\cos\omega t+b\sin\omega t$.

          Sub:
          \begin{align*}
             & (a\cos\omega t+b\sin\omega t)''
            +3(a\cos\omega t+b\sin\omega t)'                          \\
             & \qquad+2(a\cos\omega t+b\sin\omega t)=F_0\cos\omega t.
          \end{align*}
          \begin{align*}
             & (2a+3b\omega-a\omega^2)\cos\omega t                         \\
             & \qquad+(2b-3a\omega-b\omega^2)\sin\omega t=F_0\cos\omega t.
          \end{align*}
          \[
            \implies
            \begin{cases}
              (2-\omega^2)a+3\omega b=F_0, \\
              -3\omega a+(2-\omega^2)b=0.
            \end{cases}
          \]
          \[
            \begin{bmatrix}
              2-\omega^2 & 3\omega    \\
              -3\omega   & 2-\omega^2
            \end{bmatrix}
            \begin{bmatrix}
              a \\b
            \end{bmatrix}
            =
            \begin{bmatrix}
              F_0 \\0
            \end{bmatrix}
            .
          \]
          \begin{align*}
            x_p
             & =\frac{F_0}{(2-\omega^2)^2+9\omega^2}
            \bigl((2-\omega^2)\cos\omega t+3\omega\sin\omega t\bigr) \\
             & =\frac{F_0}{\sqrt{(2-\omega^2)^2+9\omega^2}}
            \left(
            \frac{2-\omega^2}{\sqrt{(2-\omega^2)^2+9\omega^2}}\cos\omega t
            +\frac{3\omega}{\sqrt{(2-\omega^2)^2+9\omega^2}}\sin\omega t
            \right)                                                  \\
             & =\frac{F_0}{\sqrt{(2-\omega^2)^2+9\omega^2}}
            \cos(\omega t-\gamma)\qquad\text{for some $\gamma$.}
          \end{align*}
  \end{enumerate}
\end{example}
